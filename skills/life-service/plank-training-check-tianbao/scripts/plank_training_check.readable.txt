# -*- coding: utf-8 -*-
"""平板支撑姿势标准与剂量核对工具。

用法:
  python plank_training_check.py --file case.json [--json]
  python plank_training_check.py --selftest

核心判定原则: 姿势不合格时, 单次时长再长也不计入成绩。
判定顺序: 先看真实缺陷(塌腰、撅臀、肘位不对、核心没收紧、憋气、
疼痛、单次超长), 再看信息是否完整; 只填了姿势却没填时长时,
姿势问题仍判不合格而不是待确认。
"""
import argparse
import io
import json
import os
import sys
import tempfile

HOLD_MIN = 20.0
HOLD_MAX = 60.0
HOLD_BAD = 120.0
HOLD_SHORT = 10.0
SETS_MIN = 2.0
SETS_MAX = 4.0
REST_MIN = 30.0

POSTURE_ITEMS = [
    ("elbow_under_shoulder", u"肘在肩正下方", False,
     u"肘部跑到肩前方会让肩关节承担大部分负荷，是肩痛的常见来源"),
    ("body_line", u"头肩髋膝踝成一条直线", False,
     u"身体不成直线说明核心与下肢支撑没有连成整体"),
    ("no_hip_sag", u"无塌腰（腰部下沉）", False,
     u"塌腰说明核心已经失守，负荷全由腰椎承担，应立即结束该组"),
    ("no_hip_up", u"无撅臀（臀部抬高）", False,
     u"撅臀时核心几乎不参与，等于白做"),
    ("core_engaged", u"核心收紧", False,
     u"核心没收紧的动作失去意义，先在仰卧位建立腹部收紧感"),
    ("head_neutral", u"头颈中立", True,
     u"头颈下垂或抬起会改变躯干力线"),
    ("feet_stable", u"脚跟后蹬、脚位稳定", True,
     u"脚尖点地不稳说明下肢支撑不充分"),
]

PAIN_ITEMS = [
    ("low_back", u"腰部疼痛或向腿部放射"),
    ("shoulder", u"肩部疼痛或抬手受限"),
    ("abdomen", u"腹部牵拉痛或不适"),
    ("dizzy", u"头晕、胸闷、眼前发黑"),
]


def num(v):
    if v is None or v == "":
        return None
    try:
        return float(v)
    except (TypeError, ValueError):
        return None


def load_json(path):
    with io.open(path, "r", encoding="utf-8-sig") as f:
        return json.load(f)


def posture_verdict(posture):
    """返回 (不合格项, 需改进项, 未填项, 姿势是否整体合格)。"""
    if not posture:
        return [], [], [name for _, name, _, _ in POSTURE_ITEMS], None
    bad = []
    warn = []
    unknown = []
    for key, name, is_warn, why in POSTURE_ITEMS:
        v = posture.get(key)
        if v is None:
            unknown.append(name)
        elif v is False:
            if is_warn:
                warn.append((name, why))
            else:
                bad.append((name, why))
    ok = (not bad) and (not unknown)
    return bad, warn, unknown, ok


def hold_verdict(hold, posture_ok):
    if hold is None:
        return u"待确认", u"未提供单次支撑时长"
    v = float(hold)
    if posture_ok is False:
        return u"不合格", u"单次 %g 秒：姿势不合格，该组时长不计入成绩，练的是代偿而不是核心" % v
    if v > HOLD_BAD:
        return u"不合格", u"单次 %g 秒，超过 %g 秒：绝大多数人此时姿势已变形，应做进阶动作而不是拉长时长" % (
            v, HOLD_BAD)
    if v < HOLD_SHORT:
        return u"需改进", u"单次 %g 秒偏短，可先退阶到跪姿或上斜版本" % v
    if v < HOLD_MIN:
        return u"需改进", u"单次 %g 秒，目标向 %g-%g 秒推进" % (v, HOLD_MIN, HOLD_MAX)
    if v > HOLD_MAX:
        return u"需改进", u"单次 %g 秒超过进阶区间，建议改抬手或抬脚等进阶动作" % v
    return u"合格", u"单次 %g 秒，在 %g-%g 秒区间" % (v, HOLD_MIN, HOLD_MAX)


def sets_verdict(sets):
    if sets is None:
        return u"待确认", u"未提供组数"
    v = float(sets)
    if v < SETS_MIN:
        return u"需改进", u"只做 %g 组，建议 %g-%g 组" % (v, SETS_MIN, SETS_MAX)
    if v > SETS_MAX:
        return u"需改进", u"做 %g 组偏多，注意总训练量" % v
    return u"合格", u"%g 组，在 %g-%g 组区间" % (v, SETS_MIN, SETS_MAX)


def rest_verdict(rest):
    if rest is None:
        return u"待确认", u"未提供组间休息时长"
    v = float(rest)
    if v < REST_MIN:
        return u"需改进", u"组间休息 %g 秒偏短，建议 30-60 秒" % v
    if v > 120.0:
        return u"需改进", u"组间休息 %g 秒偏长，可压到 30-60 秒" % v
    return u"合格", u"组间休息 %g 秒，在合理区间" % v


def build_report(data):
    posture = data.get("posture", {}) or {}
    hold = num(data.get("hold_seconds"))
    sets = num(data.get("sets"))
    rest = num(data.get("rest_sec"))
    breath = data.get("breath_holding")
    pain = data.get("pain", {}) or {}

    bad, warn, unknown, posture_ok = posture_verdict(posture)
    pain_hit = [name for key, name in PAIN_ITEMS if pain.get(key) is True]

    defects = []
    fine = []
    pending = []

    for name, why in bad:
        defects.append(u"姿势不合格：%s —— %s" % (name, why))
    for name in unknown:
        pending.append(u"姿势项未填：%s" % name)

    if pain_hit:
        defects.append(u"存在不适（%s）：停止练习并就医评估，不要靠调整姿势硬撑" % u"、".join(pain_hit))
    if breath is True:
        defects.append(u"憋气：会升高血压、加快疲劳，还会让核心更难维持，全程正常呼吸")

    hv, hm = hold_verdict(hold, posture_ok)
    if hv == u"不合格":
        defects.append(hm)
    elif hv == u"需改进":
        fine.append(hm)
    elif hv == u"待确认":
        pending.append(hm)

    sv, sm = sets_verdict(sets)
    if sv == u"需改进":
        fine.append(sm)
    elif sv == u"待确认":
        pending.append(sm)

    rv, rm = rest_verdict(rest)
    if rv == u"需改进":
        fine.append(rm)
    elif rv == u"待确认":
        pending.append(rm)

    for name, why in warn:
        fine.append(u"姿势需改进：%s —— %s" % (name, why))

    counted = (hold is not None and posture_ok is True and hv == u"合格")

    lines = []
    lines.append("=" * 62)
    lines.append(u"平板支撑姿势与剂量核对")
    lines.append("=" * 62)
    lines.append(u"单次时长：%s 秒    组数：%s    组间休息：%s 秒" % (
        (u"%g" % hold) if hold is not None else u"--",
        (u"%g" % sets) if sets is not None else u"--",
        (u"%g" % rest) if rest is not None else u"--"))
    lines.append("")
    lines.append(u"[一] 姿势判定")
    if posture_ok is None:
        lines.append(u"  未填任何姿势项，无法判定")
    else:
        lines.append(u"  姿势整体：%s" % (u"合格" if posture_ok else u"不合格"))
        for name, why in bad:
            lines.append(u"    不合格：%s —— %s" % (name, why))
        for name, why in warn:
            lines.append(u"    需改进：%s —— %s" % (name, why))
    lines.append(u"  呼吸：%s" % (u"憋气（不合格）" if breath is True else u"未见憋气"))
    lines.append("")
    lines.append(u"[二] 时长与组数")
    lines.append(u"  时长判定：%s —— %s" % (hv, hm))
    lines.append(u"  组数判定：%s —— %s" % (sv, sm))
    lines.append(u"  休息判定：%s —— %s" % (rv, rm))
    lines.append(u"  本次是否计入成绩：%s" % (u"是" if counted else u"否"))
    lines.append("")
    lines.append(u"[三] 不合格项与原因")
    if defects:
        for i, d in enumerate(defects, 1):
            lines.append(u"  %d. %s" % (i, d))
    else:
        lines.append(u"  无")
    lines.append("")
    lines.append(u"[四] 需改进项")
    if fine:
        for i, d in enumerate(fine, 1):
            lines.append(u"  %d. %s" % (i, d))
    else:
        lines.append(u"  无")
    lines.append("")
    lines.append(u"[五] 待确认项")
    if pending:
        for i, d in enumerate(pending, 1):
            lines.append(u"  %d. %s" % (i, d))
    else:
        lines.append(u"  无")
    lines.append("")
    lines.append(u"[六] 处方")
    lines.append(u"  1. 肘在肩正下方，前臂与肩同宽，头肩髋膝踝成一条直线")
    if any(n == u"无塌腰（腰部下沉）" for n, _ in bad):
        lines.append(u"  2. 塌腰：立即结束该组，改跪姿或上斜版本重建核心张力")
    elif any(n == u"无撅臀（臀部抬高）" for n, _ in bad):
        lines.append(u"  2. 撅臀：把臀降到与肩同高，缩短时长")
    else:
        lines.append(u"  2. 核心收紧：肚脐向脊柱方向收，不是把腰往下压")
    lines.append(u"  3. 全程正常呼吸，用 3-4 秒一吸、3-4 秒一呼的慢节奏")
    lines.append(u"  4. 剂量：标准姿势 3 组 × 20-40 秒，组间休息 30-60 秒")
    lines.append(u"  5. 撑不到 20 秒用跪姿或上斜；能标准撑满 60 秒再考虑抬手抬脚")
    lines.append(u"  6. 姿势变形那一刻就结束该组，该组时长不计入成绩")

    if defects:
        overall = u"不合格"
    elif fine:
        overall = u"需改进"
    elif pending:
        overall = u"待确认"
    else:
        overall = u"合格"

    lines.append("")
    lines.append(u"总判定：%s" % overall)
    lines.append(u"提示：腰腿放射痛、肩部夜间痛、腹部牵拉痛、头晕胸闷时立即停止并就医。")
    lines.append(u"免责：本结果为姿势自查输出，不构成医学诊断或康复方案。")
    lines.append("=" * 62)

    summary = {
        "posture_ok": posture_ok,
        "posture_bad": [n for n, _ in bad],
        "hold_verdict": hv,
        "counted": counted,
        "defects": defects,
        "fine": fine,
        "pending": pending,
        "overall": overall,
    }
    return "\n".join(lines), summary


def _run_checks():
    r = []
    b, w, u, ok = posture_verdict({})
    r.append((u"未填姿势时返回 None", ok is None and len(u) == len(POSTURE_ITEMS)))
    b1, _, _, ok1 = posture_verdict({
        "elbow_under_shoulder": True, "body_line": True, "no_hip_sag": True,
        "no_hip_up": True, "core_engaged": True, "head_neutral": True, "feet_stable": True})
    r.append((u"全合格姿势判合格", ok1 is True and not b1))
    b2, _, _, ok2 = posture_verdict({
        "elbow_under_shoulder": True, "body_line": True, "no_hip_sag": False,
        "no_hip_up": True, "core_engaged": True, "head_neutral": True, "feet_stable": True})
    r.append((u"反例：塌腰判姿势不合格", ok2 is False))
    r.append((u"反例：塌腰进不合格项", any(u"塌腰" in n for n, _ in b2)))
    b3, w3, _, ok3 = posture_verdict({
        "elbow_under_shoulder": True, "body_line": True, "no_hip_sag": True,
        "no_hip_up": True, "core_engaged": True, "head_neutral": False, "feet_stable": True})
    r.append((u"头颈不正只判需改进", ok3 is True and len(w3) == 1))

    r.append((u"反例：姿势不合格时长不计入", hold_verdict(300, False)[0] == u"不合格"))
    r.append((u"反例：单次 180 秒判不合格", hold_verdict(180, True)[0] == u"不合格"))
    r.append((u"单次 30 秒判合格", hold_verdict(30, True)[0] == u"合格"))
    r.append((u"反例：单次 5 秒判需改进", hold_verdict(5, True)[0] == u"需改进"))
    r.append((u"反例：只做 1 组判需改进", sets_verdict(1)[0] == u"需改进"))
    r.append((u"反例：组间休息 10 秒判需改进", rest_verdict(10)[0] == u"需改进"))

    _, s = build_report({
        "hold_seconds": 30, "sets": 3, "rest_sec": 45, "breath_holding": False,
        "posture": {"elbow_under_shoulder": True, "body_line": True, "no_hip_sag": True,
                    "no_hip_up": True, "core_engaged": True, "head_neutral": True,
                    "feet_stable": True},
        "pain": {},
    })
    r.append((u"标准样例判定合格", s["overall"] == u"合格"))
    r.append((u"标准样例时长计入成绩", s["counted"] is True))

    _, s1 = build_report({
        "hold_seconds": 300, "sets": 3, "rest_sec": 45, "breath_holding": True,
        "posture": {"elbow_under_shoulder": True, "body_line": True, "no_hip_sag": False,
                    "no_hip_up": True, "core_engaged": True, "head_neutral": True,
                    "feet_stable": True},
        "pain": {},
    })
    r.append((u"反例：塌腰加憋气加超长判不合格", s1["overall"] == u"不合格"))
    r.append((u"反例：塌腰进缺陷表", any(u"塌腰" in d for d in s1["defects"])))
    r.append((u"反例：超长时长不计入成绩", s1["counted"] is False))

    _, s2 = build_report({
        "hold_seconds": 30, "sets": 3, "rest_sec": 45,
        "posture": {"elbow_under_shoulder": True, "body_line": True, "no_hip_sag": True,
                    "no_hip_up": True, "core_engaged": True, "head_neutral": True,
                    "feet_stable": True},
        "pain": {"low_back": True},
    })
    r.append((u"反例：腰痛判不合格并提示停止", s2["overall"] == u"不合格"))
    r.append((u"反例：腰痛进缺陷表", any(u"腰部疼痛" in d for d in s2["defects"])))

    _, s3 = build_report({"posture": {"no_hip_sag": False}})
    r.append((u"缺时长但有塌腰不得判待确认", s3["overall"] == u"不合格"))

    _, s4 = build_report({})
    r.append((u"全未填记为待确认", s4["overall"] == u"待确认"))

    p = os.path.join(tempfile.gettempdir(), "_plank_bom_case.json")
    blob = b"\xef\xbb\xbf" + json.dumps({
        "hold_seconds": 30, "sets": 3, "rest_sec": 45, "breath_holding": False,
        "posture": {"no_hip_sag": True},
    }, ensure_ascii=False).encode("utf-8")
    with open(p, "wb") as f:
        f.write(blob)
    try:
        d = load_json(p)
        r.append((u"UTF-8 BOM 输入首键可读", abs(d.get("hold_seconds", 0) - 30.0) < 1e-9))
        r.append((u"UTF-8 BOM 输入嵌套结构可读", d.get("posture", {}).get("no_hip_sag") is True))
    finally:
        if os.path.exists(p):
            os.remove(p)

    return r


def selftest():
    results = _run_checks()
    failed = [name for name, ok in results if not ok]
    ok = len(failed) == 0
    lines = ["SELFTEST %s" % ("OK" if ok else "FAILED")]
    lines.append("checks=%d passed=%d failed=%d" % (len(results), len(results) - len(failed), len(failed)))
    for name in failed:
        lines.append("FAILED: %s" % name)
    with io.open(os.path.join(os.path.dirname(os.path.abspath(__file__)), "_selftest_result.txt"),
                 "w", encoding="utf-8") as f:
        f.write("\n".join(lines))
    print("\n".join(lines))
    return ok


def main():
    parser = argparse.ArgumentParser(description=u"平板支撑姿势与剂量核对工具")
    parser.add_argument("--file", help=u"输入 JSON 文件路径（UTF-8）")
    parser.add_argument("--json", action="store_true", help=u"以 JSON 输出")
    parser.add_argument("--out", help=u"结果写入指定文件")
    parser.add_argument("--selftest", action="store_true", help=u"运行自检")
    args = parser.parse_args()

    here = os.path.dirname(os.path.abspath(__file__))

    if args.selftest:
        sys.exit(0 if selftest() else 1)

    if not args.file:
        parser.error(u"需要 --file 或 --selftest")

    data = load_json(args.file)

    report, summary = build_report(data)
    body = json.dumps(summary, ensure_ascii=False, indent=2) if args.json else report
    out_path = args.out or os.path.join(here, "_last_result.txt")
    with io.open(out_path, "w", encoding="utf-8") as f:
        f.write(body)
    print(body)
    print("")
    print(u"结果已写入：%s" % out_path)


if __name__ == "__main__":
    main()
