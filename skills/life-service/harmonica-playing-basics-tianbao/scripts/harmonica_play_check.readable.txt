"""口琴吹奏基础核查：C 调十孔口琴吹/吸音孔对照、旋律逐步音孔与吹吸一致性核查、单音与气息基础提示。

用法:
  python harmonica_play_check.py --file melody.json [--out result.md] [--format json]
  python harmonica_play_check.py --selftest
"""
import argparse
import json
import math
import os
import sys

    sys.stdout.reconfigure(encoding='utf-8')
except Exception:
    pass

# C 调十孔口琴：孔号 -> 吹/吸 音名
BLOW = {1: 'C4', 2: 'E4', 3: 'G4', 4: 'C5', 5: 'E5', 6: 'G5', 7: 'C6', 8: 'E6', 9: 'G6', 10: 'C7'}
DRAW = {1: 'D4', 2: 'G4', 3: 'B4', 4: 'D5', 5: 'F5', 6: 'A5', 7: 'B5', 8: 'D6', 9: 'F6', 10: 'A6'}

NOTE_MIDI = {
    'C4': 60, 'D4': 62, 'E4': 64, 'F4': 65, 'G4': 67, 'A4': 69, 'B4': 71,
    'C5': 72, 'D5': 74, 'E5': 76, 'F5': 77, 'G5': 79, 'A5': 81, 'B5': 83,
    'C6': 84, 'D6': 86, 'E6': 88, 'F6': 89, 'G6': 91, 'A6': 93,     'C7': 96,
}


def norm(s):
    return str(s or '').strip().lower().replace(' ', '')




NOTE_MIDI_N = {norm(k): v for k, v in NOTE_MIDI.items()}


def freq_of(note):
    m = NOTE_MIDI_N.get(norm(note))
    if m is None:
        return None
    return 440.0 * (2.0 ** ((m - 69.0) / 12.0))


def note_at(hole, breath):
    """返回该孔该吹吸方式对应的音名，无效返回 None。"""
    if not isinstance(hole, int):
        try:
            hole = int(hole)
        except (TypeError, ValueError):
            return None
    if hole not in BLOW:
        return None
    b = norm(breath)
    if b == 'blow' or b == '吹':
        return BLOW[hole]
    if b == 'draw' or b == '吸':
        return DRAW[hole]
    return None


def step_check(step):
    """单步核查。step: {'hole': 1, 'breath': 'blow', 'note': 'C4'}。"""
    if not isinstance(step, dict):
        raise ValueError('sequence 的每一项必须是对象')
    hole = step.get('hole')
    breath = step.get('breath')
    note = norm(step.get('note'))
    if hole is None or breath is None or not note:
        raise ValueError('每一步需要 hole / breath / note')
    want = note_at(hole, breath)
    if want is None:
        # 区分：孔无效 vs 吹吸方式无效
        try:
            h = int(hole)
        except (TypeError, ValueError):
            h = None
        if h not in BLOW:
            return {'ok': False, 'sev': 3, 'hole': hole, 'breath': breath, 'note': note,
                    'detail': '孔 %s 无效（应为 1 到 10）' % hole,
                    'correct': None, 'why': '孔号超出十孔范围，无法发声'}
        return {'ok': False, 'sev': 3, 'hole': hole, 'breath': breath, 'note': note,
                'detail': '孔 %s 的吹吸方式「%s」无效（应为 blow 或 draw）' % (hole, breath),
                'correct': None, 'why': '吹吸方式写错，脚本无法判定音'}
    if norm(want) != note:
        # 给出正确孔：找 blow/draw 等于 note 的孔
        correct = find_hole(note)
        return {'ok': False, 'sev': 2, 'hole': hole, 'breath': breath, 'note': note,
                'detail': '%s 孔%s 应为 %s，实为 %s'
                % (hole, breath, want, note),
                'correct': correct,
                'why': '孔或吹吸弄错，吹出来就是别的音；正确应为 %s' % correct}
    return {'ok': True, 'sev': 0, 'hole': hole, 'breath': breath, 'note': note,
            'detail': '', 'correct': None, 'why': ''}


def find_hole(note):
    """返回「孔 吹/吸」字符串，找不到返回 None。按音名（忽略八度）匹配最低音孔。"""
    n = norm(note).rstrip('0123456789')
    for h in range(1, 11):
        if norm(BLOW[h]).rstrip('0123456789') == n:
            return '%d 孔吹' % h
        if norm(DRAW[h]).rstrip('0123456789') == n:
            return '%d 孔吸' % h
    return None


def calc(data):
    seq = data.get('sequence')
    if seq is None:
        raise ValueError('输入缺少 sequence 列表')
    if not isinstance(seq, list) or not seq:
        raise ValueError('sequence 必须是非空列表')
    steps = [step_check(s) for s in seq]
    issues = []
    for i, st in enumerate(steps):
        if not st['ok']:
            issues.append({'sev': st['sev'], 'item': '第 %d 步' % (i + 1),
                           'detail': st['detail'],
                           'why': st['why'], 'correct': st.get('correct')})
    issues.sort(key=lambda r: -r['sev'])
    n_err = len([s for s in steps if not s['ok']])
    if any(i['sev'] == 3 for i in issues):
        level = '有明显问题'
    elif n_err:
        level = '有吹奏错误'
    else:
        level = '合格'
    return {'level': level, 'steps': steps, 'issues': issues,
            'n_err': n_err, 'blow': BLOW, 'draw': DRAW}


def render_md(r):
    L = []
    L.append('# 口琴吹奏核查')
    L.append('')
    L.append('## 一、总体判定')
    L.append('')
    L.append('**%s**' % r['level'])
    L.append('')
    L.append('## 二、逐音孔判定')
    L.append('')
    L.append('| 步 | 孔 | 吹吸 | 目标音 | 结果 |')
    L.append('|---|---|---|---|---|')
    for i, st in enumerate(r['steps']):
        L.append('| %d | %s | %s | %s | %s |'
                 % (i + 1, st.get('hole'), st.get('breath'), st.get('note'),
                    '正确' if st['ok'] else '错误'))
    L.append('')
    if r['issues']:
        L.append('## 三、错误与整改')
        L.append('')
        L.append('| 严重度 | 步骤 | 问题 | 为什么要在意 |')
        L.append('|---|---|---|---|')
        for it in r['issues']:
            L.append('| %s | %s | %s | %s |'
                     % ('严重' if it['sev'] == 3 else '可改进', it['item'], it['detail'], it['why']))
        L.append('')
    L.append('## 四、人工确认清单')
    L.append('')
    L.append('- [ ] 已背下十孔吹吸对照')
    L.append('- [ ] 单音能吹干净不破音')
    L.append('- [ ] 吹吸用腹部气息不鼓腮')
    L.append('- [ ] 小星星能逐孔找准')
    return '\n'.join(L)


def selftest():
    errs = []
    n = 0

    def ck(cond, msg):
        nonlocal n
        n += 1
        if not cond:
            errs.append(msg)

    # 1 吹音频率
    ck(abs(freq_of('C4') - 261.63) < 0.01, 'C4 应为 261.63')
    ck(abs(freq_of('E4') - 329.63) < 0.01, 'E4 应为 329.63')
    ck(abs(freq_of('G4') - 392.00) < 0.01, 'G4 应为 392.00')
    ck(abs(freq_of('C5') - 523.25) < 0.01, 'C5 应为 523.25')
    ck(abs(freq_of('C7') - 2093.0) < 0.01, 'C7 应为 2093.00')

    # 2 吸音频率
    ck(abs(freq_of('D4') - 293.66) < 0.01, 'D4 应为 293.66')
    ck(abs(freq_of('B4') - 493.88) < 0.01, 'B4 应为 493.88')
    ck(abs(freq_of('F5') - 698.46) < 0.01, 'F5 应为 698.46')
    ck(abs(freq_of('A5') - 880.00) < 0.01, 'A5 应为 880.00')

    # 3 note_at 对照
    ck(note_at(1, 'blow') == 'C4', '1孔吹应为 C4')
    ck(note_at(1, 'draw') == 'D4', '1孔吸应为 D4')
    ck(note_at(7, 'blow') == 'C6', '7孔吹应为 C6')
    ck(note_at(6, 'draw') == 'A5', '6孔吸应为 A5')
    ck(note_at(11, 'blow') is None, '11孔应无效')
    ck(note_at(1, 'both') is None, 'both 吹吸应无效')

    # 4 全正确正例
    seq_ok = [{'hole': 1, 'breath': 'blow', 'note': 'C4'},
              {'hole': 6, 'breath': 'blow', 'note': 'G5'},
              {'hole': 6, 'breath': 'draw', 'note': 'A5'}]
    r4 = calc({'sequence': seq_ok})
    ck(r4['level'] == '合格', '全正确应合格，实为 %s' % r4['level'])
    ck(r4['n_err'] == 0, '全正确应 0 错')

    # 5 反例：1孔吹说成 D4（吹吸反了）
    r5 = calc({'sequence': [{'hole': 1, 'breath': 'blow', 'note': 'D4'}]})
    ck(r5['level'] == '有吹奏错误', '1孔吹D4应判错误，实为 %s' % r5['level'])
    ck(r5['issues'][0]['detail'].find('C4') >= 0, '应给出正确音 C4，实为 %s' % r5['issues'][0]['detail'])
    ck(r5['issues'][0]['correct'] == '1 孔吸', '应指出 1 孔吸，实为 %s' % r5['issues'][0]['correct'])

    # 6 反例：4孔吹说成 E4（孔错）
    r6 = calc({'sequence': [{'hole': 4, 'breath': 'blow', 'note': 'E4'}]})
    ck(any(i['item'] == '第 1 步' for i in r6['issues']), '应标记第1步')
    ck(r6['issues'][0]['correct'].find('2 孔') >= 0, '应指出 2 孔吹，实为 %s' % r6['issues'][0]['correct'])

    # 7 反例：无效孔号
    r7 = calc({'sequence': [{'hole': 11, 'breath': 'blow', 'note': 'C4'}]})
    ck(r7['level'] == '有明显问题', '无效孔应判严重，实为 %s' % r7['level'])
    ck(r7['issues'][0]['sev'] == 3, '无效孔应 sev3')

    # 8 反例：多步错误都报出
    r8 = calc({'sequence': [{'hole': 1, 'breath': 'blow', 'note': 'D4'},
                            {'hole': 4, 'breath': 'blow', 'note': 'E4'}]})
    ck(r8['n_err'] == 2, '应报 2 个错，实为 %d' % r8['n_err'])

    # 9 小星星映射
    star = ['C4', 'C4', 'G4', 'G4', 'A4', 'A4', 'G4', 'F4', 'F4', 'E4', 'E4', 'D4', 'D4', 'C4']
    mapped = [find_hole(s) for s in star]
    ck(mapped[0] == '1 孔吹', '小星星首音应为 1 孔吹，实为 %s' % mapped[0])
    ck(mapped[6] == '2 孔吸', '小星星第7音 G 应为最低有效孔 2 孔吸，实为 %s' % mapped[6])
    ck(all(m is not None for m in mapped), '小星星所有音都应能在琴上找到')

    # 10 判定顺序：无效孔不应因信息完整判轻
    r10 = calc({'sequence': [{'hole': 11, 'breath': 'blow', 'note': 'C4'}]})
    ck(r10['level'] == '有明显问题', '无效孔不应判轻，实为 %s' % r10['level'])

    # 11 非法输入
    for bad, why in (
        ({}, '缺 sequence'),
        ({'sequence': []}, 'sequence 空'),
        ({'sequence': [{'hole': 1}]}, '缺 breath/note'),
        ({'sequence': [{'hole': 1, 'breath': 'blow'}]}, '缺 note'),
        ({'sequence': 'x'}, 'sequence 非列表'),
    ):
        try:
            calc(bad)
            ck(False, '用例11 %s 应抛错' % why)
        except ValueError:
            ck(True, '')

    # 12 BOM 两条路径
    bom = '﻿' + json.dumps({'sequence': [{'hole': 1, 'breath': 'blow', 'note': 'D4'}]})
    r12 = calc(json.loads(bom.lstrip('﻿')))
    ck(r12['level'] == '有吹奏错误', 'BOM 字符串应解析，实为 %s' % r12['level'])
    tmp = os.path.join(os.path.dirname(os.path.abspath(__file__)), '_bom_probe_h.json')
    with open(tmp, 'w', encoding='utf-8-sig') as f:
        f.write(json.dumps({'sequence': [{'hole': 4, 'breath': 'blow', 'note': 'E4'}]}))
    try:
        r12b = calc(json.loads(open(tmp, 'r', encoding='utf-8-sig').read()))
        ck(r12b['n_err'] == 1, 'BOM 文件应解析出 1 个错，实为 %s' % r12b['n_err'])
    finally:
        if os.path.exists(tmp):
            os.remove(tmp)

    # 13 输出结构
    md = render_md(r5)
    for key in ('总体判定', '逐音孔判定', '错误与整改', '人工确认清单'):
        ck(key in md, 'Markdown 应含章节 %s' % key)
    ck(len(BLOW) == 10 and len(DRAW) == 10, '吹吸应各 10 个音')

    out = os.path.join(os.path.dirname(os.path.abspath(__file__)), '_selftest_result.txt')
    msg = 'SELFTEST FAIL\n' + '\n'.join(errs) if errs else (
        'SELFTEST OK\n断言数: %d\n覆盖: 吹/吸音频率/十孔对照/全正确正例/'
        '1孔吹反例/4孔错孔反例/无效孔反例/多步错误/小星星映射/判定顺序/'
        '非法输入/BOM两条路径/输出结构' % n)
    with open(out, 'w', encoding='utf-8') as f:
        f.write(msg + '\n')
    print(msg)
    return 0 if not errs else 1


def main():
    ap = argparse.ArgumentParser(description='口琴吹奏基础核查')
    ap.add_argument('--file', help='输入 JSON 文件')
    ap.add_argument('--text', help='输入 JSON 字符串')
    ap.add_argument('--out', help='结果输出路径')
    ap.add_argument('--format', default='md', choices=['md', 'json'])
    ap.add_argument('--selftest', action='store_true')
    a = ap.parse_args()

    if a.selftest:
        return selftest()

    raw = a.text or ''
    if a.file:
        if not os.path.exists(a.file):
            print('ERROR --file 不存在: %s' % a.file)
            return 2
        raw = open(a.file, 'r', encoding='utf-8-sig', errors='replace').read()
    if not raw.strip():
        print('ERROR 需要 --file 或 --text')
        return 2
    try:
        data = json.loads(raw.lstrip('﻿'))
    except ValueError as e:
        print('ERROR JSON 解析失败: %s' % e)
        return 2
    try:
        r = calc(data)
    except ValueError as e:
        print('ERROR %s' % e)
        return 2

    out = render_md(r) if a.format == 'md' else json.dumps(r, ensure_ascii=False, indent=2)
    d = os.path.dirname(os.path.abspath(__file__))
    target = a.out or os.path.join(d, '_last_result.txt')
    with open(target, 'w', encoding='utf-8') as f:
        f.write(out + '\n')
    print('结果已写入: %s' % target)
    print(out)
    return 0


if __name__ == '__main__':
    sys.exit(main())
