def norm(k):
    return str(k or '').strip().lower().replace(' ', '')


def knife_of(key):
    k = norm(key)
    if k not in KNIVES:
        raise ValueError('未知刀型键: %s，可选键见 KNIVES 表' % key)
    return k, KNIVES[k]


def angle_tier(a):
    if a <= 15:
        return '薄刃（适合精细切割，易崩需谨慎）'
    if a <= 20:
        return '标准刃（兼顾锋利与耐用）'
    return '厚刃（耐用，适合劈砍与重活）'


def angle_valid(a):
    try:
        v = float(a)
    except (TypeError, ValueError):
        return False
    return 0 < v <= 30


def calc(data):
    items = data.get('items') or []
    checks = data.get('checks') or {}
    if not isinstance(checks, dict):
        raise ValueError('checks 必须是对象')
    for k in checks:
        if not any(norm(c[0]) == norm(k) for c in CHECK_ITEMS):
            raise ValueError('未知的检查项: %s，可选项见 CHECK_ITEMS 表' % k)

    if items:
        rows = []
        for it in items:
            key, (name, angle, tool, grit, note) = knife_of(it.get('knife'))
            use_angle = it.get('angle')
            if use_angle is not None and not angle_valid(use_angle):
                raise ValueError('angle 必须在 0 到 30 之间，实为: %s' % use_angle)
            rows.append({
                'name': it.get('name', name),
                'knife': key,
                'knife_name': name,
                'rec_angle': angle,
                'use_angle': use_angle,
                'angle_tier': angle_tier(angle),
                'tool': tool,
                'grit': grit,
                'note': note,
                'serrated': key in ('bread_knife', 'serrated_bread', 'kitchen_scissors'),
                'wood_handle': key in ('chinese_cleaver', 'beef_knife', 'chopping_hachet'),
            })
        rows.sort(key=lambda r: r['knife_name'])
    else:
        rows = []

    ck_rows = []
    bad = []
    unk = []
    for ckey, cname, cdesc, cwhy in CHECK_ITEMS:
        v = checks.get(ckey)
        if v is None:
            ck_rows.append({'name': cname, 'status': '未检查', 'desc': cdesc, 'why': cwhy})
            unk.append(cname)
        elif v:
            ck_rows.append({'name': cname, 'status': '合格', 'desc': cdesc, 'why': cwhy})
        else:
            ck_rows.append({'name': cname, 'status': '不合格', 'desc': cdesc, 'why': cwhy})
            bad.append(cname)

    warns = []
    for r in rows:
        if r['serrated']:
            warns.append('%s 属锯齿或剪刀类，**不可磨成平刃**，否则失去锯切能力' % r['name'])
        if r['wood_handle']:
            warns.append('%s 为木质柄类，**不可进洗碗机、不可长时间泡水**，'
                         '否则柄部开裂变形且刀身锈蚀' % r['name'])
        if r['use_angle'] is not None and abs(float(r['use_angle']) - r['rec_angle']) >= 5:
            warns.append('%s 建议刃角 %d 度，当前按 %s 度，偏差较大，刃线几何会改变'
                         % (r['name'], r['rec_angle'], r['use_angle']))
    for b in bad:
        warns.append('检查不合格：%s，%s' % (b, next(c[3] for c in CHECK_ITEMS if c[1] == b)))
    for u in unk:
        warns.append('检查未完成：%s，建议逐项确认后再继续使用' % u)

    if len(bad) >= 1:
        level = '需处理后使用'
    elif unk:
        level = '待确认'
    else:
        level = '状态良好'

    return {
        'level': level,
        'items': rows,
        'check_rows': ck_rows,
        'bad': bad,
        'unchecked': unk,
        'warns': warns,
        'tool_compare': [{'name': a, 'good': b, 'bad': c, 'use': d}
                         for a, b, c, d in TOOL_COMPARE],
        'grind_rule': GRIND_RULE,
        'burrs_note': BURRS_NOTE,
        'storage_rule': STORAGE_RULE,
        'banned': BANNED,
        'separation': COLOR_SEPARATION,
    }

def selftest():
    errs = []
    n = 0

    def ck(cond, msg):
        nonlocal n
        n += 1
        if not cond:
            errs.append(msg)

    # 1 刃角档位
    ck(angle_tier(15) == '薄刃（适合精细切割，易崩需谨慎）', '15 度档位有误: %s' % angle_tier(15))
    ck(angle_tier(20) == '标准刃（兼顾锋利与耐用）', '20 度档位有误: %s' % angle_tier(20))
    ck(angle_tier(25) == '厚刃（耐用，适合劈砍与重活）', '25 度档位有误: %s' % angle_tier(25))

    # 2 刃角合法性
    ck(angle_valid(15) is True, '15 度应合法')
    ck(angle_valid(0) is False, '0 度应非法')
    ck(angle_valid(31) is False, '31 度应非法')

    # 3 中式片刀参数
    r3 = calc({'items': [{'knife': 'chinese_cleaver'}]})
    ck(r3['items'][0]['rec_angle'] == 15, '中式片刀应为 15 度，实为 %s' % r3['items'][0]['rec_angle'])
    ck(r3['items'][0]['wood_handle'] is True, '中式片刀应标木柄类')
    ck(any('洗碗机' in w for w in r3['warns']), '木柄类应提示不可进洗碗机，实为 %s' % r3['warns'])

    # 4 日式主厨刀
    r4 = calc({'items': [{'knife': 'japanese_gyuto'}]})
    ck(r4['items'][0]['knife_name'] == '日式主厨刀', '刀型名有误: %s' % r4['items'][0]['knife_name'])
    ck('只磨主刃' in r4['items'][0]['note'], '日式刀应说明只磨主刃，实为 %s' % r4['items'][0]['note'])

    # 5 锯齿刀不可磨平
    r5 = calc({'items': [{'knife': 'serrated_bread'}]})
    ck(r5['items'][0]['serrated'] is True, '锯齿刀应标锯齿类')
    ck(any('不可磨成平刃' in w for w in r5['warns']), '锯齿刀应提示不可磨平，实为 %s' % r5['warns'])

    # 6 剪刀不可整体上磨刀器
    ck(any('剪刀不可整体压上磨刀器' in b for b in BANNED), '必须含剪刀不可整体上磨刀器')
    ck(any('磨刀棒只能修整不能磨出锋利刃口' in b for b in BANNED), '必须含磨刀棒的限制说明')

    # 7 磨刀方向口径
    ck('逆刃推磨' in GRIND_RULE, '必须含逆刃推磨')
    ck('不是来回拉锯' in GRIND_RULE, '必须明确不是来回拉锯')
    ck('由刀跟向刀尖' in GRIND_RULE, '必须含单向推动方向')

    # 8 去毛刺要求
    ck('去毛刺' in BURRS_NOTE, '必须含去毛刺要求')
    ck('挂毛' in BURRS_NOTE, '去毛刺应说明挂毛后果')

    # 9 收纳四条
    ck('不可叠放' in STORAGE_RULE, '收纳必须含不可叠放')
    ck('刃口不碰金属' in STORAGE_RULE, '收纳必须含刃口不碰金属')
    ck('竖放' in STORAGE_RULE, '收纳必须含竖放或刀架')

    # 10 检查项三级判定
    allok = {k: True for k, _, _, _ in CHECK_ITEMS}
    r10 = calc({'checks': allok})
    ck(r10['level'] == '状态良好', '全合格应状态良好，实为 %s' % r10['level'])
    ck(len(r10['check_rows']) == 9, '检查项应 9 条，实为 %d' % len(r10['check_rows']))
    r10b = calc({'checks': dict(allok, burr=False)})
    ck(r10b['level'] == '需处理后使用', '毛刺不合格应需处理，实为 %s' % r10b['level'])
    ck('毛刺已清除' in r10b['bad'], '应点名毛刺项，实为 %s' % r10b['bad'])
    r10c = calc({'checks': {k: True for k, _, _, _ in CHECK_ITEMS if k != 'corrosion'}})
    ck(r10c['level'] == '待确认', '缺项应待确认，实为 %s' % r10c['level'])
    ck('无锈蚀' in r10c['unchecked'], '应列出未确认的无锈蚀，实为 %s' % r10c['unchecked'])

    # 11 角度偏差告警
    r11 = calc({'items': [{'knife': 'japanese_gyuto', 'angle': 25}]})
    ck(any('偏差较大' in w for w in r11['warns']), '角度偏差应告警，实为 %s' % r11['warns'])

    # 12 生熟分离要求
    ck('生熟刀必须分离' in COLOR_SEPARATION, '必须含生熟刀分离')
    ck('沙门氏菌' in COLOR_SEPARATION, '生熟分离应说明致病菌')

    # 13 BOM 两条路径
    bom = '﻿{"items": [{"knife": "boning_knife"}]}'
    r13 = calc(json.loads(bom.lstrip('﻿')))
    ck(r13['items'][0]['rec_angle'] == 20, 'BOM 输入应解析，剔骨刀应为 20 度，实为 %s' % r13['items'][0]['rec_angle'])
    tmp = os.path.join(os.path.dirname(os.path.abspath(__file__)), '_bom_probe6.json')
    with open(tmp, 'w', encoding='utf-8-sig') as f:
        f.write(json.dumps({'checks': dict(allok, storage=False)}))
    try:
        r13b = calc(json.loads(open(tmp, 'r', encoding='utf-8-sig').read()))
        ck(r13b['level'] == '需处理后使用', 'utf-8-sig 读取应正常，实为 %s' % r13b['level'])
        ck('收纳合规' in r13b['bad'], '应点名收纳项，实为 %s' % r13b['bad'])
    finally:
        if os.path.exists(tmp):
            os.remove(tmp)

    # 14 非法输入
    for bad, why in (
        ({'items': [{'knife': '不存在的刀'}]}, '未知刀型'),
        ({'checks': {'不存在的项': True}}, '未知检查项'),
        ({'checks': '不是对象'}, 'checks 非对象'),
        ({'items': [{'knife': 'chinese_cleaver', 'angle': 45}]}, '角度超范围'),
    ):
        try:
            calc(bad)
            ck(False, '用例14 %s 应抛错' % why)
        except ValueError:
            ck(True, '')

    # 15 输出结构
    md = render_md(r3)
    for key in ('磨刀器与磨刀石', '逆刃推磨', '去毛刺', '收纳', '禁止', '生熟刀分离', '人工确认清单'):
        ck(key in md, 'Markdown 输出应含 %s' % key)
    ck(len(r3['banned']) >= 7, '禁止项至少 7 条，实为 %d' % len(r3['banned']))

    out = os.path.join(os.path.dirname(os.path.abspath(__file__)), '_selftest_result.txt')
    msg = 'SELFTEST FAIL\n' + '\n'.join(errs) if errs else (
        'SELFTEST OK\n断言数: %d\n覆盖: 刃角档位/刃角合法性/中式片刀木柄禁忌/日式单刃/锯齿不可磨平/'
        '剪刀与磨刀棒限制/逆刃推磨口径/去毛刺/收纳四条/检查项三级判定/角度偏差告警/'
        '生熟分离/BOM 两条路径/非法输入/输出结构' % n)
    with open(out, 'w', encoding='utf-8') as f:
        f.write(msg + '\n')
    print(msg)
    return 0 if not errs else 1


def main():
    ap = argparse.ArgumentParser(description='刀具保养与磨刀核查')
    ap.add_argument('--file', help='输入 JSON 文件')
    ap.add_argument('--text', help='输入 JSON 字符串')
    ap.add_argument('--out', help='结果输出路径')
    ap.add_argument('--format', default='md', choices=['md', 'json'])
    ap.add_argument('--selftest', action='store_true')
    a = ap.parse_args()

    if a.selftest:
        return selftest()

    raw = a.text or ''
    if a.file:
        if not os.path.exists(a.file):
            print('ERROR --file 不存在: %s' % a.file)
            return 2
        raw = open(a.file, 'r', encoding='utf-8-sig', errors='replace').read()
    if not raw.strip():
        print('ERROR 需要 --file 或 --text')
        return 2
    try:
        data = json.loads(raw.lstrip('﻿'))
    except ValueError as e:
        print('ERROR JSON 解析失败: %s' % e)
        return 2
    try:
        r = calc(data)
    except ValueError as e:
        print('ERROR %s' % e)
        return 2

    out = render_md(r) if a.format == 'md' else json.dumps(r, ensure_ascii=False, indent=2)
    d = os.path.dirname(os.path.abspath(__file__))
    target = a.out or os.path.join(d, '_last_result.txt')
    with open(target, 'w', encoding='utf-8') as f:
        f.write(out + '\n')
    print('结果已写入: %s' % target)
    print(out)
    return 0


if __name__ == '__main__':
    sys.exit(main())
