# 食材: (键, 中文名, 卤制下限分钟, 卤制上限分钟, 浸泡下限分钟, 备注)
ITEMS = [
    ('poultry_part', '鸡翅、鸡爪', 15, 25, 30, '爪类可延长时间更软糯'),
    ('whole_chicken', '整鸡、半鸡', 30, 45, 30, '关火后连同汤一起冷却更好'),
    ('duck_part', '鸭脖、鸭头、鸭掌', 20, 30, 30, '腥味重，焯水要彻底'),
    ('pork_trotter', '猪蹄、猪肘', 60, 90, 60, '胶质重，卤水冷却会结冻属正常'),
    ('beef_shank', '牛腱', 60, 90, 60, '必须彻底冷却再切，否则散碎'),
    ('pork_belly', '五花肉、后腿肉', 30, 45, 30, '先煎一下表面更香并定型'),
    ('egg', '蛋类', 8, 12, 240, '剥壳后下锅，泡得越久色越深'),
    ('tofu', '豆制品（豆干、素鸡）', 5, 10, 20, '极易吸咸，时间短也不要提高含盐量'),
    ('root_veg', '根茎类（藕、土豆、海带）', 3, 8, 10, '素菜单锅原则'),
]

ITEM_MAP = dict((r[0], r) for r in ITEMS)

# 温度纪律（摄氏度）
TEMP = {'min_ok': 85.0, 'max_ok': 95.0, 'low_fail': 80.0}

# 含盐量区间
SALT = {'min': 1.5, 'max': 2.5}

# 存放与复热口径
STORAGE = {'room_hours_max': 2.0, 'fridge_hours_max': 48.0, 'reheat_min': 70.0}

# 变质四项信号: (键, 中文名, 说明)
SPOILAGE = [
    ('white_film', '表面起白膜', '意味着产膜酵母已经成群落'),
    ('ropy', '拉丝', '挑起来能拉丝'),
    ('sour', '酸味或馊味', '出现酸味、馊味、腐败味'),
    ('viscous', '发黏', '卤水触感黏滑、挂壁发稠'),
]

# 六条明确禁止的操作
BANNED = [
    '大火滚煮卤制（肉柴、卤水乳化发浑发酸、咸度失控）',
    '卤完不浸泡直接切（外面咸里面淡）',
    '老卤使用前不煮沸',
    '素菜尤其是豆制品进主锅，且用完的卤水倒回老卤',
    '变质卤水撇掉表层继续用',
    '使用任何硝盐、防腐剂、护色剂',
]

LEVEL_DISCARD = '必须弃用'
LEVEL_HARD = '必须处理'
LEVEL_WEAK = '存在薄弱环节'
LEVEL_UNKNOWN = '待确认'
LEVEL_OK = '操作规范'


def num(v, name):
    if v is None:
        return None
    try:
        return float(v)
    except (TypeError, ValueError):
        raise ValueError('%s 必须是数字' % name)


def get_item(kind):
    key = str(kind or '').strip().lower()
    if key not in ITEM_MAP:
        raise ValueError('未知的食材种类: %s，可选 %s 之一' % (kind, '/'.join(ITEM_MAP.keys())))
    return ITEM_MAP[key]


def check_items(items):
    if items is None:
        items = []
    if not isinstance(items, list):
        raise ValueError('items 必须是列表')
    rows = []
    hard = []
    weak = []
    for raw in items:
        if not isinstance(raw, dict):
            raise ValueError('items 的每一项必须是对象')
        kind = raw.get('kind')
        row = get_item(kind)
        key, name, bmin, bmax, smin, note = row
        bm = num(raw.get('braise_minutes'), 'braise_minutes')
        sm = num(raw.get('soak_minutes'), 'soak_minutes')
        label = raw.get('name') or name
        rec = {'name': label, 'kind_name': name, 'note': note,
               'braise': bm, 'braise_range': '%d 至 %d' % (bmin, bmax),
               'soak': sm, 'soak_min': smin}
        if bm is None or sm is None:
            rec['status'] = '未检查'
            rec['why'] = '缺少卤制分钟或浸泡分钟，无法判定'
            rows.append(rec)
            continue
        problems = []
        if bm < bmin:
            problems.append('卤制 %g 分钟低于该类食材下限 %d 分钟，内部未熟透' % (bm, bmin))
        elif bm > bmax:
            problems.append('卤制 %g 分钟高于该类食材上限 %d 分钟，肉质容易散烂' % (bm, bmax))
        # 浸泡下限: 不低于卤制时间的一半, 同时不低于该类食材的浸泡下限
        need = max(smin, bm / 2.0)
        if sm < need:
            problems.append('浸泡 %g 分钟不足（需要至少 %g 分钟），成品只会表层咸、里面淡' % (sm, need))
        if problems:
            rec['status'] = '不合格'
            rec['why'] = '；'.join(problems)
            rows.append(rec)
            if sm < bm / 2.0 or bm < bmin:
                hard.append(label)
            else:
                weak.append(label)
        else:
            rec['status'] = '合格'
            rec['why'] = '卤制与浸泡都符合该类食材的口径'
            rows.append(rec)
    return rows, hard, weak


def check_spoilage(flags):
    if flags is None:
        flags = {}
    if not isinstance(flags, dict):
        raise ValueError('spoilage 必须是对象')
    keys = [s[0] for s in SPOILAGE]
    for k in flags:
        if str(k).strip().lower() not in keys:
            raise ValueError('未知的变质信号: %s，可选项见 SPOILAGE 表' % k)
    hit = []
    rows = []
    unk = []
    for key, name, desc in SPOILAGE:
        v = flags.get(key)
        if v is True:
            hit.append(name)
            rows.append({'name': name, 'status': '出现', 'desc': desc})
        elif v is False:
            rows.append({'name': name, 'status': '未出现', 'desc': desc})
        else:
            rows.append({'name': name, 'status': '未检查', 'desc': desc})
            unk.append(name)
    return rows, hit, unk


def calc(data):
    if not isinstance(data, dict):
        raise ValueError('输入必须是 JSON 对象')

    brine = data.get('brine') or {}
    if not isinstance(brine, dict):
        raise ValueError('brine 必须是对象')

    rows, hard_items, weak_items = check_items(data.get('items'))
    spoil_rows, spoil_hit, spoil_unk = check_spoilage(data.get('spoilage'))

    # 温度核查
    pot = num(data.get('pot_temp_c'), 'pot_temp_c')
    rolling = data.get('rolling_boil')
    temp_row = None
    temp_hard = False
    if pot is not None:
        if rolling is True:
            status = '不合格'
            why = '大火滚煮会让肌肉纤维收缩变柴、脂肪乳化使卤水发浑发酸、汤汁蒸发导致咸度失控'
            temp_hard = True
        elif pot > TEMP['max_ok']:
            status = '偏高'
            why = '高于 95 摄氏度，接近沸腾，应立即转最小火保持虾眼水'
        elif pot < TEMP['low_fail']:
            status = '不合格'
            why = '低于 80 摄氏度，保温不足，入味慢且变质风险上升'
            temp_hard = True
        elif pot < TEMP['min_ok']:
            status = '偏低'
            why = '低于 85 摄氏度目标下限，可略微调大火到 85 至 95 摄氏度区间'
        else:
            status = '达标'
            why = '落在 85 至 95 摄氏度的微沸区间'
        temp_row = {'value': pot, 'status': status, 'why': why,
                    'range': '%.0f 至 %.0f' % (TEMP['min_ok'], TEMP['max_ok'])}

    # 含盐量核查
    sal = num(brine.get('salinity_pct'), 'salinity_pct')
    salt_row = None
    salt_bad = salt_missing = False
    if sal is not None:
        if sal < SALT['min']:
            status = '不合格'
            why = '低于 1.5，入味不足且抑菌能力下降，常温存放风险上升'
            salt_bad = True
        elif sal > SALT['max']:
            status = '不合格'
            why = '高于 2.5，过咸且会掩盖卤水开始变质的酸味，让人错过判废窗口'
            salt_bad = True
        else:
            status = '达标'
            why = '落在 1.5 至 2.5 区间，按卤水总重计并已折算酱油带入的盐'
        salt_row = {'value': sal, 'status': status, 'why': why}
    else:
        salt_missing = True

    # 养护动作
    care_rows = []
    care_hard = []
    care_weak = []
    bo = brine.get('boiled_before_use')
    if bo is None:
        care_rows.append({'name': '老卤用前煮沸不少于 5 分钟', 'status': '未检查',
                          'why': '冷藏只是延缓不是杀菌'})
    elif bo:
        care_rows.append({'name': '老卤用前煮沸不少于 5 分钟', 'status': '合格',
                          'why': '既杀菌也唤醒香味'})
    else:
        care_rows.append({'name': '老卤用前煮沸不少于 5 分钟', 'status': '不合格',
                          'why': '冷藏只是延缓不是杀菌，取出直接使用是安全缺陷'})
        care_hard.append('老卤用前未煮沸')
    fi = brine.get('filtered_after_use')
    if fi is None:
        care_rows.append({'name': '用后过滤与煮沸', 'status': '未检查',
                          'why': '残渣是下次酸败的种子'})
    elif fi:
        care_rows.append({'name': '用后过滤与煮沸', 'status': '合格',
                          'why': '已完成捞出、细筛、煮沸流程'})
    else:
        care_rows.append({'name': '用后过滤与煮沸', 'status': '不合格',
                          'why': '未过滤残渣，下一次酸败的种子已留在锅里'})
        care_weak.append('用后未过滤')

    # 素菜单锅
    veg_row = None
    veg_hard = False
    veg = data.get('veg_in_main_pot')
    if veg is None:
        veg_row = {'status': '未检查', 'why': '需要确认本次是否放了豆制品、根茎或菌菇'}
    elif veg is not True and veg is not False:
        raise ValueError('veg_in_main_pot 必须是布尔值或留空')
    elif veg:
        veg_row = {'status': '不合格',
                   'why': '素菜带入大量水分稀释卤水，豆制品的蛋白质与糖分是杂菌培养基，是老卤酸败的头号成因'}
        veg_hard = True
    else:
        veg_row = {'status': '合格', 'why': '素菜单独起小锅，用完的卤水未倒回'}

    # 存放与复热
    cool = num(data.get('cool_hours'), 'cool_hours')
    cool_row = None
    cool_hard = False
    if cool is None:
        cool_row = {'status': '未检查', 'why': '室温放置时长未填'}
    elif cool > STORAGE['room_hours_max']:
        cool_row = {'value': cool, 'status': '不合格',
                    'why': '室温放置 %g 小时超过 2 小时上限，4 至 60 摄氏度是细菌快速繁殖区间' % cool}
        cool_hard = True
    else:
        cool_row = {'value': cool, 'status': '合格', 'why': '在 2 小时内进入冷藏'}

    stored = num(data.get('stored_hours'), 'stored_hours')
    stored_row = None
    stored_hard = False
    if stored is None:
        stored_row = {'status': '未检查', 'why': '冷藏存放时长未填'}
    elif stored > STORAGE['fridge_hours_max']:
        stored_row = {'value': stored, 'status': '不合格',
                      'why': '冷藏 %g 小时超过 48 小时上限，风味与安全性都不可控' % stored}
        stored_hard = True
    else:
        stored_row = {'value': stored, 'status': '合格', 'why': '在 48 小时以内'}

    reheat = num(data.get('reheat_temp'), 'reheat_temp')
    reheat_row = None
    reheat_hard = False
    if reheat is None:
        reheat_row = {'status': '未检查', 'why': '复热温度未填'}
    elif reheat < STORAGE['reheat_min']:
        reheat_row = {'value': reheat, 'status': '不合格',
                      'why': '复热到 %g 摄氏度，低于 70 摄氏度中心温度要求，等于只加热了表面' % reheat}
        reheat_hard = True
    else:
        reheat_row = {'value': reheat, 'status': '合格', 'why': '达到 70 摄氏度以上'}

    # 判定: 变质信号最高优先, 其次食品安全类缺陷, 再次口味与时间项,
    # 最后才看信息是否完整。
    if spoil_hit:
        level = LEVEL_DISCARD
    elif (temp_hard or care_hard or veg_hard or cool_hard or stored_hard
          or reheat_hard or (salt_bad and brine.get('storage') != 'fridge')):
        level = LEVEL_HARD
    elif hard_items or salt_bad:
        level = LEVEL_HARD
    elif weak_items or care_weak or (temp_row and temp_row['status'] == '偏高'):
        level = LEVEL_WEAK
    elif (temp_row is None or spoil_unk
          or veg_row['status'] == '未检查'
          or any(x['status'] == '未检查' for x in (cool_row, stored_row, reheat_row))
          or any(x['status'] == '未检查' for x in rows)
          or any(x['status'] == '未检查' for x in care_rows)
          or salt_missing):
        level = LEVEL_UNKNOWN
    else:
        level = LEVEL_OK

    return {
        'items_rows': rows,
        'hard_items': hard_items,
        'weak_items': weak_items,
        'temp_row': temp_row,
        'salt_row': salt_row,
        'salt_missing': salt_missing,
        'care_rows': care_rows,
        'veg_row': veg_row,
        'cool_row': cool_row,
        'stored_row': stored_row,
        'reheat_row': reheat_row,
        'spoil_rows': spoil_rows,
        'spoil_hit': spoil_hit,
        'spoil_unk': spoil_unk,
        'level': level,
        'limits': {'temp': TEMP, 'salt': SALT, 'storage': STORAGE},
    }


def render_md(r):
    L = []
    L.append('# 卤味制作与卤水养护核查结果')
    L.append('')
    L.append('**总体判定：%s**' % r['level'])
    L.append('')
    L.append('## 一、变质信号（最高优先级）')
    L.append('')
    L.append('| 信号 | 状态 | 说明 |')
    L.append('|---|---|---|')
    for row in r['spoil_rows']:
        L.append('| %s | %s | %s |' % (row['name'], row['status'], row['desc']))
    if r['spoil_hit']:
        L.append('')
        L.append('> 出现「%s」，**整锅弃用**。掏掉表层继续用不算处理，霉菌代谢产物已扩散到整锅。'
                 % '、'.join(r['spoil_hit']))
    L.append('')
    L.append('## 二、食材时间与浸泡')
    L.append('')
    if not r['items_rows']:
        L.append('未提供食材清单，无法核查时间与浸泡。')
    else:
        L.append('| 食材 | 种类 | 卤制（分钟） | 推荐区间 | 浸泡（分钟） | 浸泡下限 | 状态 |')
        L.append('|---|---|---|---|---|---|---|')
        for row in r['items_rows']:
            L.append('| %s | %s | %s | %s | %s | %d | %s |' % (
                row['name'], row['kind_name'],
                ('%g' % row['braise']) if row['braise'] is not None else '未填',
                row['braise_range'],
                ('%g' % row['soak']) if row['soak'] is not None else '未填',
                row['soak_min'], row['status']))
        bad = [x for x in r['items_rows'] if x['status'] == '不合格']
        if bad:
            L.append('')
            L.append('不合格项及原因：')
            for row in bad:
                L.append('- **%s**：%s' % (row['name'], row['why']))
    L.append('')
    L.append('## 三、温度纪律')
    L.append('')
    if r['temp_row'] is None:
        L.append('未提供锅温数据，无法核查。目标区间 85 至 95 摄氏度，禁止大火滚煮。')
    else:
        t = r['temp_row']
        L.append('| 实测（摄氏度） | 目标区间 | 状态 | 说明 |')
        L.append('|---|---|---|---|')
        L.append('| %.1f | %s | %s | %s |' % (t['value'], t['range'], t['status'], t['why']))
    L.append('')
    L.append('## 四、卤水含盐量')
    L.append('')
    if r['salt_row'] is None:
        L.append('未提供含盐量，无法核查。建议口径 1.5 至 2.5（按卤水总重，含酱油折算）。')
    else:
        s = r['salt_row']
        L.append('| 实测 | 建议区间 | 状态 | 说明 |')
        L.append('|---|---|---|---|')
        L.append('| %.2f | 1.5 至 2.5 | %s | %s |' % (s['value'], s['status'], s['why']))
    L.append('')
    L.append('## 五、老卤养护与素菜单锅')
    L.append('')
    L.append('| 项目 | 状态 | 说明 |')
    L.append('|---|---|---|')
    for row in r['care_rows']:
        L.append('| %s | %s | %s |' % (row['name'], row['status'], row['why']))
    L.append('| 素菜单锅原则 | %s | %s |' % (r['veg_row']['status'], r['veg_row']['why']))
    L.append('')
    L.append('## 六、存放与复热')
    L.append('')
    L.append('| 项目 | 状态 | 说明 |')
    L.append('|---|---|---|')
    L.append('| 室温放置 | %s | %s |' % (r['cool_row']['status'], r['cool_row']['why']))
    L.append('| 冷藏存放 | %s | %s |' % (r['stored_row']['status'], r['stored_row']['why']))
    L.append('| 复热温度 | %s | %s |' % (r['reheat_row']['status'], r['reheat_row']['why']))
    L.append('')
    L.append('口径：室温不超过 2 小时、冷藏不超过 48 小时、复热中心温度不低于 70 摄氏度。')
    L.append('')
    L.append('## 七、六条明确禁止的操作')
    L.append('')
    for i, b in enumerate(BANNED, 1):
        L.append('%d. %s' % (i, b))
    L.append('')
    L.append('## 八、人工确认清单')
    L.append('')
    L.append('- [ ] 卤水含盐量在 1.5 至 2.5 之间（含酱油折算）')
    L.append('- [ ] 卤制全程保持 85 至 95 摄氏度，没有大火滚煮')
    L.append('- [ ] 每种食材的浸泡时间不少于其卤制时间的一半')
    L.append('- [ ] 老卤这次使用前煮沸了不少于 5 分钟')
    L.append('- [ ] 用后已完成过滤、煮沸、撇油')
    L.append('- [ ] 素菜（尤其豆制品）没有进主锅')
    L.append('- [ ] 成品出锅后 2 小时内进入冷藏')
    L.append('- [ ] 冷藏不超过 48 小时，复热到中心温度 70 摄氏度以上')
    L.append('- [ ] 尝过一遍：没有酸、馊、黏滑，表面没有白膜')
    L.append('')
    L.append('## 九、免责')
    L.append('')
    L.append('本结果为家庭厨房方法论整理，不检测亚硝酸盐与致病菌，不构成食品安全判定结论。')
    return '\n'.join(L)


def selftest():
    errs = []
    n = 0

    def ck(cond, msg):
        nonlocal n
        n += 1
        if not cond:
            errs.append(msg)

    base = {
        'brine': {'salinity_pct': 1.8, 'boiled_before_use': True,
                  'filtered_after_use': True, 'storage': 'fridge', 'age_days': 40},
        'pot_temp_c': 93,
        'rolling_boil': False,
        'items': [{'name': '牛腱', 'kind': 'beef_shank', 'braise_minutes': 70, 'soak_minutes': 90}],
        'veg_in_main_pot': False,
        'cool_hours': 1.5,
        'stored_hours': 20,
        'reheat_temp': 72,
        'spoilage': {'white_film': False, 'ropy': False, 'sour': False, 'viscous': False},
    }

    # 1 正例
    r1 = calc(base)
    ck(r1['level'] == LEVEL_OK, '正例应为操作规范，实为 %s' % r1['level'])
    ck(r1['items_rows'][0]['status'] == '合格', '牛腱应判合格，实为 %s' % r1['items_rows'][0]['status'])

    # 2 反例: 变质信号优先于一切
    r2 = calc(dict(base, spoilage={'white_film': True, 'ropy': False, 'sour': False, 'viscous': False}))
    ck(r2['level'] == LEVEL_DISCARD, '出现白膜应判必须弃用，实为 %s' % r2['level'])
    ck('表面起白膜' in r2['spoil_hit'], '应点名白膜，实为 %s' % r2['spoil_hit'])
    r2b = calc(dict(base, spoilage={'white_film': False, 'ropy': False, 'sour': False, 'viscous': True}))
    ck(r2b['level'] == LEVEL_DISCARD, '出现发黏应判必须弃用，实为 %s' % r2b['level'])

    # 3 反例: 大火滚煮
    r3 = calc(dict(base, rolling_boil=True))
    ck(r3['level'] == LEVEL_HARD, '大火滚煮应判必须处理，实为 %s' % r3['level'])
    ck('变柴' in r3['temp_row']['why'], '必须带出为什么，实为 %s' % r3['temp_row']['why'])

    # 4 反例: 温度过低
    r4 = calc(dict(base, pot_temp_c=72))
    ck(r4['temp_row']['status'] == '不合格', '低于 80 度应判不合格，实为 %s' % r4['temp_row']['status'])
    ck(r4['level'] == LEVEL_HARD, '保温不足应判必须处理，实为 %s' % r4['level'])

    # 5 反例: 卤制时间不足
    r5 = calc(dict(base, items=[{'name': '牛腱', 'kind': 'beef_shank',
                                 'braise_minutes': 25, 'soak_minutes': 90}]))
    ck(r5['items_rows'][0]['status'] == '不合格', '牛腱卤 25 分钟应判不合格')
    ck('未熟透' in r5['items_rows'][0]['why'], '时间不足必须带出原因，实为 %s' % r5['items_rows'][0]['why'])
    ck(r5['level'] == LEVEL_HARD, '时间不足应判必须处理，实为 %s' % r5['level'])

    # 6 反例: 浸泡不足(这是最常见的坑)
    r6 = calc(dict(base, items=[{'name': '牛腱', 'kind': 'beef_shank',
                                 'braise_minutes': 70, 'soak_minutes': 0}]))
    ck(r6['items_rows'][0]['status'] == '不合格', '完全不浸泡应判不合格')
    ck('里面淡' in r6['items_rows'][0]['why'], '浸泡不足必须说明只会表层咸，实为 %s' % r6['items_rows'][0]['why'])
    r6b = calc(dict(base, items=[{'name': '牛腱', 'kind': 'beef_shank',
                                  'braise_minutes': 70, 'soak_minutes': 20}]))
    ck(r6b['items_rows'][0]['status'] == '不合格', '浸泡 20 分钟少于 35 分钟应判不合格')

    # 7 反例: 含盐量偏低
    r7 = calc(dict(base, brine={'salinity_pct': 1.0, 'boiled_before_use': True,
                                'filtered_after_use': True, 'storage': 'fridge'}))
    ck(r7['salt_row']['status'] == '不合格', '含盐 1.0 应判不合格，实为 %s' % r7['salt_row']['status'])
    ck(r7['level'] == LEVEL_HARD, '含盐偏低应判必须处理，实为 %s' % r7['level'])

    # 8 反例: 含盐量偏高会掩盖酸味
    r8 = calc(dict(base, brine={'salinity_pct': 3.2, 'boiled_before_use': True,
                                'filtered_after_use': True, 'storage': 'fridge'}))
    ck(r8['salt_row']['status'] == '不合格', '含盐 3.2 应判不合格')
    ck('掩盖' in r8['salt_row']['why'], '含盐偏高必须说明会掩盖酸味，实为 %s' % r8['salt_row']['why'])

    # 9 反例: 老卤未煮沸
    r9 = calc(dict(base, brine={'salinity_pct': 1.8, 'boiled_before_use': False,
                                'filtered_after_use': True, 'storage': 'fridge'}))
    ck(r9['level'] == LEVEL_HARD, '老卤未煮沸应判必须处理，实为 %s' % r9['level'])
    ck(r9['care_rows'][0]['status'] == '不合格', '应点名老卤煮沸项，实为 %s' % r9['care_rows'][0]['status'])

    # 10 反例: 素菜进主锅
    r10 = calc(dict(base, veg_in_main_pot=True))
    ck(r10['level'] == LEVEL_HARD, '素菜进主锅应判必须处理，实为 %s' % r10['level'])
    ck('酸败' in r10['veg_row']['why'], '必须说明是酸败成因，实为 %s' % r10['veg_row']['why'])

    # 11 反例: 存放超时与复热不足
    r11 = calc(dict(base, cool_hours=4))
    ck(r11['level'] == LEVEL_HARD, '室温放置 4 小时应判必须处理，实为 %s' % r11['level'])
    r11b = calc(dict(base, stored_hours=72))
    ck(r11b['level'] == LEVEL_HARD, '冷藏 72 小时应判必须处理，实为 %s' % r11b['level'])
    r11c = calc(dict(base, reheat_temp=60))
    ck(r11c['level'] == LEVEL_HARD, '复热 60 度应判必须处理，实为 %s' % r11c['level'])

    # 12 种类表与蛋类浸泡
    ck(len(ITEMS) == 9, '食材种类应 9 类，实为 %d' % len(ITEMS))
    r12 = calc(dict(base, items=[{'name': '卤蛋', 'kind': 'egg',
                                  'braise_minutes': 10, 'soak_minutes': 300}]))
    ck(r12['items_rows'][0]['status'] == '合格', '卤蛋泡 300 分钟应判合格')
    r12b = calc(dict(base, items=[{'name': '卤蛋', 'kind': 'egg',
                                   'braise_minutes': 10, 'soak_minutes': 120}]))
    ck(r12b['items_rows'][0]['status'] == '不合格', '卤蛋泡 120 分钟应判不合格')

    # 13 判定分支顺序: 真实缺陷不能被信息不全压成待确认
    r13 = calc(dict(base, pot_temp_c=72, cool_hours=None))
    ck(r13['level'] == LEVEL_HARD, '温度不合格且信息不全仍应判必须处理，实为 %s' % r13['level'])
    r13b = calc(dict(base, reheat_temp=None, stored_hours=None))
    ck(r13b['level'] == LEVEL_UNKNOWN, '无缺陷但缺存放数据应判待确认，实为 %s' % r13b['level'])

    # 14 BOM 两条路径
    bom_text = '﻿' + json.dumps(dict(base, rolling_boil=True))
    r14 = calc(json.loads(bom_text.lstrip('﻿')))
    ck(r14['level'] == LEVEL_HARD, 'BOM 字符串应能正确解析，实为 %s' % r14['level'])
    ck(r14['temp_row']['value'] == 93.0, 'BOM 输入锅温应为 93，实为 %s' % r14['temp_row']['value'])
    tmp = os.path.join(os.path.dirname(os.path.abspath(__file__)), '_bom_probe_braise.json')
    with open(tmp, 'w', encoding='utf-8-sig') as f:
        f.write(json.dumps(dict(base, items=[{'name': '鸡爪', 'kind': 'poultry_part',
                                              'braise_minutes': 20, 'soak_minutes': 40}])))
    try:
        loaded = json.loads(open(tmp, 'r', encoding='utf-8-sig').read())
        r14b = calc(loaded)
        ck(r14b['items_rows'][0]['kind_name'] == '鸡翅、鸡爪',
           'utf-8-sig 读取应拿到鸡翅鸡爪，实为 %s' % r14b['items_rows'][0]['kind_name'])
        ck(r14b['level'] == LEVEL_OK, '鸡爪卤 20 泡 40 应判操作规范，实为 %s' % r14b['level'])
    finally:
        if os.path.exists(tmp):
            os.remove(tmp)

    # 15 非法输入必须报错
    for payload, why in (
        ({'items': [{'kind': 'unknown_kind', 'braise_minutes': 10, 'soak_minutes': 10}]}, '未知种类'),
        ({'items': '不是列表'}, '食材非列表'),
        ({'spoilage': {'拼错的信号': True}}, '未知变质信号'),
        ({'pot_temp_c': '很热'}, '温度非数字'),
        ({'brine': '不是对象'}, '卤水非对象'),
        ({'veg_in_main_pot': '不是布尔'}, '素菜字段非布尔'),
    ):
        try:
            calc(payload)
            ck(False, '用例15 %s 应抛错却通过' % why)
        except ValueError:
            ck(True, '')
        except Exception as e:
            ck(False, '用例15 %s 应抛 ValueError，实为 %s' % (why, e))

    # 16 输出结构
    md = render_md(r2)
    for key in ('总体判定', '变质信号', '食材时间与浸泡', '温度纪律', '卤水含盐量',
                '老卤养护与素菜单锅', '存放与复热', '六条明确禁止的操作', '人工确认清单', '免责'):
        ck(key in md, 'Markdown 输出应含章节 %s' % key)
    ck(len(BANNED) == 6, '禁止项应 6 条，实为 %d' % len(BANNED))
    ck(len(r1['spoil_rows']) == 4, '变质信号应 4 条，实为 %d' % len(r1['spoil_rows']))

    out = os.path.join(os.path.dirname(os.path.abspath(__file__)), '_selftest_result.txt')
    if errs:
        msg = 'SELFTEST FAIL\n' + '\n'.join(errs)
    else:
        msg = ('SELFTEST OK\n断言数: %d\n覆盖: 正例/变质四信号/大火滚煮/温度过低/卤制不足/'
               '浸泡不足/含盐偏低/含盐偏高掩盖酸味/老卤未煮沸/素菜入主锅/存放超时与复热不足/'
               '种类表与蛋类浸泡/判定分支顺序/BOM 两条路径/非法输入/输出结构' % n)
    with open(out, 'w', encoding='utf-8') as f:
        f.write(msg + '\n')
    print(msg)
    return 0 if not errs else 1


def main():
    ap = argparse.ArgumentParser(description='卤味制作与卤水养护核查')
    ap.add_argument('--file', help='输入 JSON 文件')
    ap.add_argument('--text', help='输入 JSON 字符串')
    ap.add_argument('--out', help='结果输出路径')
    ap.add_argument('--format', default='md', choices=['md', 'json'])
    ap.add_argument('--selftest', action='store_true')
    a = ap.parse_args()

    if a.selftest:
        return selftest()

    raw = a.text or ''
    if a.file:
        if not os.path.exists(a.file):
            print('ERROR --file 不存在: %s' % a.file)
            return 2
        raw = open(a.file, 'r', encoding='utf-8-sig', errors='replace').read()
    if not raw.strip():
        print('ERROR 需要 --file 或 --text')
        return 2
    try:
        data = json.loads(raw.lstrip('﻿'))
    except ValueError as e:
        print('ERROR JSON 解析失败: %s' % e)
        return 2
    try:
        r = calc(data)
    except ValueError as e:
        print('ERROR %s' % e)
        return 2

    out = render_md(r) if a.format == 'md' else json.dumps(r, ensure_ascii=False, indent=2)
    target = a.out or os.path.join(os.path.dirname(os.path.abspath(__file__)), '_last_result.txt')
    with open(target, 'w', encoding='utf-8') as f:
        f.write(out + '\n')
    print('结果已写入: %s' % target)
    return 0


if __name__ == '__main__':
    sys.exit(main())
