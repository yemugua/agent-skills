"""证件照自查：核分辨率、像素尺寸、底色、构图比例、五官着装与拍摄时间。

用法:
  python id_photo_check.py --file photo.json [--out result.md] [--format json]
  python id_photo_check.py --selftest
"""
import argparse
import json
import os
import sys

try:
    sys.stdout.reconfigure(encoding='utf-8')
except Exception:
    pass

# 常见规格: (键, 中文名, 毫米宽, 毫米高, 300dpi 像素宽, 300dpi 像素高, 允许底色, 常见用途)
SPECS = [
    ('one_inch', '一寸', 25, 35, 295, 413, ['white', 'blue', 'red'], '社保、资格报名、档案表'),
    ('two_inch', '二寸', 35, 49, 413, 579, ['white', 'blue', 'red'], '简历、资格证书、部分签证'),
    ('small_two', '小二寸', 35, 45, 413, 531, ['white', 'blue'], '部分考试报名'),
    ('passport', '护照与签证常用', 33, 48, 390, 567, ['white'], '出入境证件、多数国家签证'),
]

# 底色: (键, 中文名)
BACKGROUNDS = [
    ('white', '白底'),
    ('blue', '蓝底'),
    ('red', '红底'),
]

# 表情: (键, 中文名, 是否露齿)
EXPRESSIONS = [
    ('neutral', '中性表情，双唇自然闭合', False),
    ('slight', '微微抿嘴', False),
    ('teeth', '露齿微笑', True),
]

# 阈值
TOL = {
    'dpi_min': 300,          # 最低分辨率
    'px_tol': 3,             # 像素尺寸容差
    'face_min': 0.70,        # 头部占比下限
    'face_max': 0.80,        # 头部占比上限
    'top_min': 0.03,         # 头顶留白下限
    'top_max': 0.10,         # 头顶留白上限
    'eye_min': 0.45,         # 眼睛水平位置下限
    'eye_max': 0.65,         # 眼睛水平位置上限
    'months_max': 6,         # 拍摄时间上限（月）
    'kb_min': 10,            # 文件大小下限（KB）
    'kb_max': 300,           # 文件大小上限（KB）
}


def norm(k):
    return str(k or '').strip().lower().replace(' ', '')


def _f(shot, key):
    v = shot.get(key)
    if v is None:
        return None
    try:
        f = float(v)
    except (TypeError, ValueError):
        raise ValueError('%s 必须是数字，实为 %s' % (key, v))
    if f <= 0:
        raise ValueError('%s 必须为正数，实为 %s' % (key, v))
    return f


def _row(table, val, label):
    if val is None or val == '':
        return None
    n = norm(val)
    for r in table:
        if norm(r[0]) == n:
            return r
    raise ValueError('未知的 %s: %s，可选 %s 之一' % (label, val, '/'.join(r[0] for r in table)))


def shot_check(shot):
    if not isinstance(shot, dict):
        raise ValueError('photos 的每一项必须是对象')
    name = str(shot.get('name') or '未命名').strip()
    spec = _row(SPECS, shot.get('spec'), 'spec')
    bg = _row(BACKGROUNDS, shot.get('bg'), 'bg')
    expr = _row(EXPRESSIONS, shot.get('expression'), 'expression')
    px_w = _f(shot, 'px_w')
    px_h = _f(shot, 'px_h')
    dpi = _f(shot, 'dpi')
    face = _f(shot, 'face_height_ratio')
    top = _f(shot, 'head_top_margin')
    eye = _f(shot, 'eye_line_ratio')
    kb = _f(shot, 'file_kb')
    months = _f(shot, 'months_since')

    issues = []
    misses = []

    if spec is None:
        misses.append('规格')

    # 严重项 1：分辨率
    if dpi is None:
        misses.append('分辨率')
    elif dpi < TOL['dpi_min']:
        issues.append({
            'sev': 3, 'item': '分辨率',
            'detail': '%.0f dpi，低于 %d dpi' % (dpi, TOL['dpi_min']),
            'why': '分辨率不足会直接被受理系统退回；'
                   '从相册截图通常只有 72 dpi，必须按规格重新导出或拍摄',
        })

    # 严重项 2：像素尺寸
    if px_w is None or px_h is None:
        misses.append('像素尺寸')
    elif spec is not None:
        want_w, want_h = spec[4], spec[5]
        if abs(px_w - want_w) > TOL['px_tol'] or abs(px_h - want_h) > TOL['px_tol']:
            issues.append({
                'sev': 3, 'item': '像素尺寸',
                'detail': '当前 %.0f × %.0f，%s 要求 %d × %d'
                          % (px_w, px_h, spec[1], want_w, want_h),
                'why': '像素尺寸不符会被系统直接拒收；'
                       '像素 = 毫米 ÷ 25.4 × dpi，按受理机构公布的数值导出',
            })

    # 严重项 3：底色
    if bg is None:
        misses.append('底色')
    elif spec is not None and norm(bg[0]) not in spec[6]:
        issues.append({
            'sev': 3, 'item': '底色',
            'detail': '%s，%s 通常只接受 %s'
                      % (bg[1], spec[1], '、'.join(_bg_name(x) for x in spec[6])),
            'why': '底色不符会被受理机构退回；拍摄时就选对底色，比后期抠图换底可靠',
        })

    # 严重项 4：眉毛与耳朵
    if shot.get('hair_cover_brow'):
        issues.append({
            'sev': 3, 'item': '眉毛',
            'detail': '头发遮挡眉毛',
            'why': '眉毛必须完整露出，遮挡即被退回；刘海梳上去或用发夹固定',
        })
    if shot.get('ears_visible') is False:
        issues.append({
            'sev': 3, 'item': '耳朵',
            'detail': '耳朵未露出',
            'why': '多数用途要求露出耳朵；长发别到耳后再拍',
        })

    # 严重项 5：帽子头饰
    if shot.get('headwear'):
        issues.append({
            'sev': 3, 'item': '帽子头饰',
            'detail': '佩戴了帽子或头饰',
            'why': '证件照不得佩戴帽子头饰（宗教与医疗原因按受理机构要求办理）；'
                   '拍摄前取下帽子、发箍、夸张发饰',
        })

    # 严重项 6：拍摄时间
    if months is None:
        misses.append('拍摄时间')
    elif months > TOL['months_max']:
        issues.append({
            'sev': 3, 'item': '拍摄时间',
            'detail': '距拍摄已 %.0f 个月，超过 %d 个月' % (months, TOL['months_max']),
            'why': '证件照通常要求近 6 个月内拍摄，超期需重拍',
        })

    # 严重项 7：文件大小
    if kb is None:
        misses.append('文件大小')
    elif not TOL['kb_min'] <= kb <= TOL['kb_max']:
        why = ('小于 %d KB 通常画质不足' % TOL['kb_min'] if kb < TOL['kb_min']
               else '大于 %d KB 多数报名系统传不上去' % TOL['kb_max'])
        issues.append({
            'sev': 3, 'item': '文件大小',
            'detail': '%.0f KB，合格区间 %d 到 %d KB' % (kb, TOL['kb_min'], TOL['kb_max']),
            'why': why,
        })

    # 严重项 8：红眼
    if shot.get('red_eye'):
        issues.append({
            'sev': 3, 'item': '红眼',
            'detail': '存在红眼',
            'why': '红眼会被直接退回；关掉闪光灯，改用两盏常亮灯从左右前方补光',
        })

    # 严重项 9：服装与底色对比
    if shot.get('clothes_contrast') is False:
        issues.append({
            'sev': 3, 'item': '服装配色',
            'detail': '服装颜色与底色缺乏对比',
            'why': '白底穿白衬衫会让肩膀与背景糊在一起；换成与底色有明确保持对比的颜色',
        })

    # 可改进项：头部占比
    if face is None:
        misses.append('头部占比')
    elif not TOL['face_min'] <= face <= TOL['face_max']:
        why = ('小于 %.2f 人显得太小、画面空' % TOL['face_min'] if face < TOL['face_min']
               else '大于 %.2f 容易切到下巴或顶得太满' % TOL['face_max'])
        issues.append({
            'sev': 2, 'item': '头部占比',
            'detail': '%.2f，合格区间 %.2f 到 %.2f' % (face, TOL['face_min'], TOL['face_max']),
            'why': why,
        })

    # 可改进项：头顶留白
    if top is None:
        misses.append('头顶留白')
    elif not TOL['top_min'] <= top <= TOL['top_max']:
        why = ('小于 %.2f 顶得太满甚至被判头顶过满' % TOL['top_min'] if top < TOL['top_min']
               else '大于 %.2f 人显小、画面松散' % TOL['top_max'])
        issues.append({
            'sev': 2, 'item': '头顶留白',
            'detail': '%.2f，合格区间 %.2f 到 %.2f' % (top, TOL['top_min'], TOL['top_max']),
            'why': why,
        })

    # 可改进项：眼睛位置
    if eye is None:
        misses.append('眼睛水平位置')
    elif not TOL['eye_min'] <= eye <= TOL['eye_max']:
        issues.append({
            'sev': 2, 'item': '眼睛位置',
            'detail': '%.2f，合格区间 %.2f 到 %.2f' % (eye, TOL['eye_min'], TOL['eye_max']),
            'why': '眼睛太高像俯拍，太低显得没精神；镜头高度与眼睛齐平',
        })

    # 可改进项：背景
    if shot.get('bg_even') is False:
        issues.append({
            'sev': 2, 'item': '背景均匀度',
            'detail': '背景不均匀或有褶皱杂物',
            'why': '背景必须纯净均匀；换干净墙面或把布绷平',
        })
    if shot.get('shadow_on_bg'):
        issues.append({
            'sev': 2, 'item': '背景阴影',
            'detail': '背景上有阴影',
            'why': '人离墙太近会把影子投在墙上；离墙至少 0.5 米，'
                   '并把两盏灯移到左右前方 45 度',
        })

    # 可改进项：眼镜
    if shot.get('glasses'):
        issues.append({
            'sev': 2, 'item': '眼镜',
            'detail': '戴着眼镜',
            'why': '多数用途要求不戴眼镜；允许戴时镜片不得反光、不得遮挡眼睛、'
                   '镜框不得遮眉毛，建议直接摘掉拍',
        })

    # 可改进项：表情
    if expr is None:
        misses.append('表情')
    elif expr[2]:
        issues.append({
            'sev': 2, 'item': '表情',
            'detail': '%s' % expr[1],
            'why': '多数证件照要求中性表情、双唇自然闭合，露齿笑会被退回',
        })

    # 可改进项：肩膀
    if shot.get('shoulders_visible') is False:
        issues.append({
            'sev': 2, 'item': '肩膀',
            'detail': '肩膀未拍全或两侧不对称',
            'why': '半身证件照通常要求露出肩线且两侧对称；退远一点，'
                   '让脸部中轴对准网格竖中线',
        })

    issues.sort(key=lambda r: -r['sev'])

    if any(i['sev'] == 3 for i in issues):
        level = '存在退回风险'
    elif issues:
        level = '有可改进项'
    elif misses:
        level = '待确认'
    else:
        level = '符合要求'

    return {
        'name': name, 'level': level, 'issues': issues, 'misses': misses,
        'spec': spec[1] if spec else None, 'bg': bg[1] if bg else None,
        'px_w': px_w, 'px_h': px_h, 'dpi': dpi, 'kb': kb, 'months': months,
    }


def _bg_name(key):
    for k, n in BACKGROUNDS:
        if k == key:
            return n
    return key


def calc(data):
    photos = data.get('photos')
    if photos is None:
        raise ValueError('输入缺少 photos 列表')
    if not isinstance(photos, list) or not photos:
        raise ValueError('photos 必须是非空列表')
    rows = [shot_check(p) for p in photos]
    n_bad = len([r for r in rows if r['level'] == '存在退回风险'])
    n_mid = len([r for r in rows if r['level'] == '有可改进项'])
    n_unk = len([r for r in rows if r['level'] == '待确认'])
    if n_bad:
        overall = '存在会被退回的证件照'
    elif n_mid:
        overall = '存在可改进的证件照'
    elif n_unk:
        overall = '待确认'
    else:
        overall = '全部符合要求'

    def _sev(r):
        return max([i['sev'] for i in r['issues']], default=0)

    order = sorted(rows, key=lambda r: (-_sev(r), -len(r['issues']), r['name']))
    return {
        'overall': overall, 'rows': rows,
        'n_bad': n_bad, 'n_mid': n_mid, 'n_unk': n_unk,
        'n_ok': len(rows) - n_bad - n_mid - n_unk,
        'fix_order': [r['name'] for r in order if r['issues']],
        'specs': [{'key': k, 'name': n, 'mm': '%d x %d' % (w, h),
                   'px': '%d x %d' % (pw, ph), 'bg': [_bg_name(x) for x in bgs],
                   'use': u}
                  for k, n, w, h, pw, ph, bgs, u in SPECS],
        'backgrounds': [{'key': k, 'name': n} for k, n in BACKGROUNDS],
        'expressions': [{'key': k, 'name': n, 'teeth': t} for k, n, t in EXPRESSIONS],
        'tol': dict(TOL),
    }


def render_md(r):
    L = []
    L.append('# 证件照自查')
    L.append('')
    L.append('> 不同用途与受理机构的要求差异很大，本表为常见值，**最终以受理机构公布的要求为准**。')
    L.append('')
    L.append('## 一、总体判定')
    L.append('')
    L.append('**%s**' % r['overall'])
    L.append('')
    L.append('| 类别 | 张数 |')
    L.append('|---|---|')
    L.append('| 存在退回风险 | %d |' % r['n_bad'])
    L.append('| 可改进 | %d |' % r['n_mid'])
    L.append('| 待确认 | %d |' % r['n_unk'])
    L.append('| 符合要求 | %d |' % r['n_ok'])
    L.append('')
    if r['fix_order']:
        L.append('整改顺序（严重度高的先改）：%s' % '、'.join(r['fix_order']))
        L.append('')
    L.append('## 二、逐张判定')
    L.append('')
    for row in r['rows']:
        L.append('### %s' % row['name'])
        L.append('')
        L.append('- 判定：**%s**' % row['level'])
        if row['spec']:
            L.append('- 规格：%s' % row['spec'])
        if row['px_w'] is not None and row['px_h'] is not None:
            L.append('- 像素：%.0f × %.0f' % (row['px_w'], row['px_h']))
        if row['dpi'] is not None:
            L.append('- 分辨率：%.0f dpi' % row['dpi'])
        if row['bg']:
            L.append('- 底色：%s' % row['bg'])
        if row['kb'] is not None:
            L.append('- 文件大小：%.0f KB' % row['kb'])
        if row['months'] is not None:
            L.append('- 距拍摄：%.0f 个月' % row['months'])
        if row['issues']:
            L.append('')
            L.append('| 严重度 | 项目 | 问题 | 为什么要在意 |')
            L.append('|---|---|---|---|')
            for i in row['issues']:
                L.append('| %s | %s | %s | %s |'
                         % ('严重' if i['sev'] == 3 else '可改进', i['item'], i['detail'], i['why']))
        if row['misses']:
            L.append('')
            L.append('- 未填写：%s' % '、'.join(row['misses']))
        L.append('')
    L.append('## 三、常见规格（300 dpi）')
    L.append('')
    L.append('| 规格 | 毫米 | 像素 | 允许底色 | 常见用途 |')
    L.append('|---|---|---|---|---|')
    for s in r['specs']:
        L.append('| %s | %s | %s | %s | %s |' % (s['name'], s['mm'], s['px'],
                                                 '、'.join(s['bg']), s['use']))
    L.append('')
    L.append('## 四、阈值')
    L.append('')
    L.append('| 项目 | 阈值 |')
    L.append('|---|---|')
    L.append('| 分辨率 | 不低于 %d dpi |' % r['tol']['dpi_min'])
    L.append('| 头部长度占比 | %.2f 到 %.2f |' % (r['tol']['face_min'], r['tol']['face_max']))
    L.append('| 头顶留白 | %.2f 到 %.2f |' % (r['tol']['top_min'], r['tol']['top_max']))
    L.append('| 眼睛水平位置 | %.2f 到 %.2f |' % (r['tol']['eye_min'], r['tol']['eye_max']))
    L.append('| 拍摄时间 | 不超过 %d 个月 |' % r['tol']['months_max'])
    L.append('| 文件大小 | %d 到 %d KB |' % (r['tol']['kb_min'], r['tol']['kb_max']))
    L.append('')
    L.append('## 五、人工确认清单（脚本不替代）')
    L.append('')
    L.append('- [ ] 已核对受理机构公布的具体像素、底色与文件大小要求')
    L.append('- [ ] 关掉了美颜与滤镜')
    L.append('- [ ] 镜头高度与眼睛齐平，脸部中轴对准网格竖中线')
    L.append('- [ ] 眉毛耳朵露出，没戴帽子头饰，没戴眼镜')
    L.append('- [ ] 背景纯净均匀、无阴影，人离墙至少 0.5 米')
    L.append('- [ ] 连拍了 5 到 10 张，放大检查瞳孔清晰、无红眼')
    return '\n'.join(L)


def selftest():
    errs = []
    n = 0

    def ck(cond, msg):
        nonlocal n
        n += 1
        if not cond:
            errs.append(msg)

    ok = {'name': '社保一寸', 'spec': 'one_inch', 'px_w': 295, 'px_h': 413, 'dpi': 300,
          'bg': 'white', 'face_height_ratio': 0.75, 'head_top_margin': 0.06,
          'eye_line_ratio': 0.55, 'file_kb': 45, 'months_since': 2,
          'hair_cover_brow': False, 'ears_visible': True, 'headwear': False,
          'red_eye': False, 'clothes_contrast': True, 'bg_even': True,
          'shadow_on_bg': False, 'glasses': False, 'expression': 'neutral',
          'shoulders_visible': True}

    # 1 正例
    r1 = shot_check(ok)
    ck(r1['level'] == '符合要求', '正例应判符合要求，实为 %s' % r1['level'])
    ck(r1['issues'] == [], '正例不应有问题项，实为 %s' % r1['issues'])

    # 2 反例：分辨率不足
    r2 = shot_check(dict(ok, dpi=72))
    ck(r2['level'] == '存在退回风险', '72 dpi 应判退回风险，实为 %s' % r2['level'])
    ck(any(i['item'] == '分辨率' for i in r2['issues']), '应点名分辨率')
    ck(shot_check(dict(ok, dpi=300))['issues'] == [], '300 dpi 边界应合格')

    # 3 反例：像素尺寸不符
    r3 = shot_check(dict(ok, px_w=200, px_h=280))
    ck(any(i['item'] == '像素尺寸' for i in r3['issues']), '像素不符应判退回风险')
    ck(not any(i['item'] == '像素尺寸' for i in shot_check(dict(ok, px_w=297, px_h=415))['issues']),
       '3 像素容差内应合格')
    ck(any(i['item'] == '像素尺寸' for i in shot_check(dict(ok, px_w=299, px_h=413))['issues']),
       '超出容差应不合格')

    # 4 反例：底色不符
    r4 = shot_check(dict(ok, spec='passport', px_w=390, px_h=567, bg='blue'))
    ck(any(i['item'] == '底色' for i in r4['issues']), '护照用蓝底应判退回风险')
    ck(not any(i['item'] == '底色' for i in shot_check(dict(ok, spec='two_inch',
                                                           px_w=413, px_h=579, bg='blue'))['issues']),
       '二寸蓝底应合格')

    # 5 反例：眉毛遮挡、耳朵未露、戴帽
    ck(any(i['item'] == '眉毛' for i in shot_check(dict(ok, hair_cover_brow=True))['issues']),
       '遮眉应判退回风险')
    ck(any(i['item'] == '耳朵' for i in shot_check(dict(ok, ears_visible=False))['issues']),
       '耳朵未露应判退回风险')
    ck(any(i['item'] == '帽子头饰' for i in shot_check(dict(ok, headwear=True))['issues']),
       '戴帽应判退回风险')

    # 6 反例：超过 6 个月
    r6 = shot_check(dict(ok, months_since=8))
    ck(any(i['item'] == '拍摄时间' for i in r6['issues']), '超 8 个月应判退回风险')
    ck(shot_check(dict(ok, months_since=6))['issues'] == [], '6 个月边界应合格')

    # 7 反例：文件大小超限
    ck(any(i['item'] == '文件大小' for i in shot_check(dict(ok, file_kb=800))['issues']),
       '800 KB 应判退回风险')
    ck(any(i['item'] == '文件大小' for i in shot_check(dict(ok, file_kb=3))['issues']),
       '3 KB 应判退回风险')
    ck(shot_check(dict(ok, file_kb=10))['issues'] == [], '10 KB 边界应合格')
    ck(shot_check(dict(ok, file_kb=300))['issues'] == [], '300 KB 边界应合格')

    # 8 反例：红眼、服装无对比
    ck(any(i['item'] == '红眼' for i in shot_check(dict(ok, red_eye=True))['issues']),
       '红眼应判退回风险')
    ck(any(i['item'] == '服装配色' for i in shot_check(dict(ok, clothes_contrast=False))['issues']),
       '服装无对比应判退回风险')

    # 9 可改进项：头部占比与边界
    ck(any(i['item'] == '头部占比' for i in shot_check(dict(ok, face_height_ratio=0.5))['issues']),
       '头部占比 0.5 应提示')
    ck(shot_check(dict(ok, face_height_ratio=0.70))['issues'] == [], '0.70 边界应合格')
    ck(shot_check(dict(ok, face_height_ratio=0.80))['issues'] == [], '0.80 边界应合格')

    # 10 可改进项：头顶留白与眼睛位置
    ck(any(i['item'] == '头顶留白' for i in shot_check(dict(ok, head_top_margin=0.2))['issues']),
       '留白 0.2 应提示')
    ck(shot_check(dict(ok, head_top_margin=0.03))['issues'] == [], '0.03 边界应合格')
    ck(any(i['item'] == '眼睛位置' for i in shot_check(dict(ok, eye_line_ratio=0.8))['issues']),
       '眼睛位置 0.8 应提示')

    # 11 可改进项：眼镜、露齿、肩膀、背景
    ck(any(i['item'] == '眼镜' for i in shot_check(dict(ok, glasses=True))['issues']),
       '戴眼镜应提示')
    ck(any(i['item'] == '表情' for i in shot_check(dict(ok, expression='teeth'))['issues']),
       '露齿笑应提示')
    ck(any(i['item'] == '肩膀' for i in shot_check(dict(ok, shoulders_visible=False))['issues']),
       '肩膀未拍全应提示')
    ck(any(i['item'] == '背景阴影' for i in shot_check(dict(ok, shadow_on_bg=True))['issues']),
       '背景有阴影应提示')
    ck(any(i['item'] == '背景均匀度' for i in shot_check(dict(ok, bg_even=False))['issues']),
       '背景不均匀应提示')

    # 12 排序方向：严重项排最前
    r12 = shot_check(dict(ok, dpi=72, glasses=True, head_top_margin=0.3))
    ck(r12['issues'][0]['sev'] == 3, '严重项应排最前，实为 %s' % r12['issues'][0]['sev'])
    ck(r12['issues'][0]['item'] == '分辨率', '首个应是分辨率，实为 %s' % r12['issues'][0]['item'])
    ck(r12['issues'][-1]['sev'] == 2, '末位应是可改进项')

    # 13 判定顺序：有严重项时不因缺信息判待确认
    r13 = shot_check({'name': '只填分辨率', 'spec': 'one_inch', 'dpi': 72})
    ck(r13['level'] == '存在退回风险', '缺其他项但有严重项不应判待确认，实为 %s' % r13['level'])
    r13b = shot_check(dict(ok, months_since=None, head_top_margin=None))
    ck(r13b['level'] == '待确认', '缺非必填项应判待确认，实为 %s' % r13b['level'])

    # 14 汇总与整改顺序
    r14 = calc({'photos': [ok,
                           dict(ok, name='低分辨率', dpi=72),
                           dict(ok, name='戴眼镜', glasses=True)]})
    ck(r14['overall'] == '存在会被退回的证件照', '有严重项应判会被退回，实为 %s' % r14['overall'])
    ck(r14['n_bad'] == 1, '严重项应为 1 张，实为 %d' % r14['n_bad'])
    ck(r14['fix_order'][0] == '低分辨率', '严重项应先改，实为 %s' % r14['fix_order'])
    ck(r14['fix_order'][1] == '戴眼镜', '可改进项排其后，实为 %s' % r14['fix_order'])

    # 15 BOM 两条路径
    bom = '﻿' + json.dumps({'photos': [dict(ok, dpi=72)]})
    r15 = calc(json.loads(bom.lstrip('﻿')))
    ck(r15['overall'] == '存在会被退回的证件照', 'BOM 字符串应正确解析，实为 %s' % r15['overall'])
    tmp = os.path.join(os.path.dirname(os.path.abspath(__file__)), '_bom_probe_c5.json')
    with open(tmp, 'w', encoding='utf-8-sig') as f:
        f.write(json.dumps({'photos': [dict(ok, name='带BOM', months_since=12)]}))
    try:
        r15b = calc(json.loads(open(tmp, 'r', encoding='utf-8-sig').read()))
        ck(r15b['rows'][0]['name'] == '带BOM', 'BOM 文件首个 key 应解析正常，实为 %s' % r15b['rows'][0]['name'])
        ck(r15b['n_bad'] == 1, '超 12 个月应判退回风险，实为 %d' % r15b['n_bad'])
    finally:
        if os.path.exists(tmp):
            os.remove(tmp)

    # 16 非法输入
    for bad, why in (
        ({}, '缺 photos'),
        ({'photos': []}, 'photos 空列表'),
        ({'photos': [dict(ok, dpi='abc')]}, 'dpi 非数字'),
        ({'photos': [dict(ok, px_w=0)]}, '像素为 0'),
        ({'photos': [dict(ok, spec='三寸')]}, '未知规格'),
        ({'photos': [dict(ok, bg='绿底')]}, '未知底色'),
        ({'photos': [dict(ok, expression='大笑')]}, '未知表情'),
    ):
        try:
            calc(bad)
            ck(False, '用例16 %s 应抛错' % why)
        except ValueError:
            ck(True, '')

    # 17 输出结构
    md = render_md(r14)
    for key in ('总体判定', '逐张判定', '常见规格', '阈值', '人工确认清单', '以受理机构公布的要求为准'):
        ck(key in md, 'Markdown 输出应含 %s' % key)
    ck(len(SPECS) == 4, '规格应 4 种')
    ck(all(s[4] > 0 and s[5] > 0 for s in SPECS), '每种规格都要有像素值')

    out = os.path.join(os.path.dirname(os.path.abspath(__file__)), '_selftest_result.txt')
    msg = 'SELFTEST FAIL\n' + '\n'.join(errs) if errs else (
        'SELFTEST OK\n断言数: %d\n覆盖: 正例/分辨率反例与边界/像素尺寸反例与容差/底色反例/'
        '遮眉与耳朵与戴帽反例/超期反例与边界/文件上下限与边界/红眼与服装反例/头部占比边界/'
        '留白与眼睛位置/眼镜表情肩膀背景/严重项排序/判定顺序/汇总整改顺序/BOM两条路径/非法输入/输出结构' % n)
    with open(out, 'w', encoding='utf-8') as f:
        f.write(msg + '\n')
    print(msg)
    return 0 if not errs else 1


def main():
    ap = argparse.ArgumentParser(description='证件照自查')
    ap.add_argument('--file', help='输入 JSON 文件')
    ap.add_argument('--text', help='输入 JSON 字符串')
    ap.add_argument('--out', help='结果输出路径')
    ap.add_argument('--format', default='md', choices=['md', 'json'])
    ap.add_argument('--selftest', action='store_true')
    a = ap.parse_args()

    if a.selftest:
        return selftest()

    raw = a.text or ''
    if a.file:
        if not os.path.exists(a.file):
            print('ERROR --file 不存在: %s' % a.file)
            return 2
        raw = open(a.file, 'r', encoding='utf-8-sig', errors='replace').read()
    if not raw.strip():
        print('ERROR 需要 --file 或 --text')
        return 2
    try:
        data = json.loads(raw.lstrip('﻿'))
    except ValueError as e:
        print('ERROR JSON 解析失败: %s' % e)
        return 2
    try:
        r = calc(data)
    except ValueError as e:
        print('ERROR %s' % e)
        return 2

    out = render_md(r) if a.format == 'md' else json.dumps(r, ensure_ascii=False, indent=2)
    d = os.path.dirname(os.path.abspath(__file__))
    target = a.out or os.path.join(d, '_last_result.txt')
    with open(target, 'w', encoding='utf-8') as f:
        f.write(out + '\n')
    print('结果已写入: %s' % target)
    print(out)
    return 0


if __name__ == '__main__':
    sys.exit(main())
