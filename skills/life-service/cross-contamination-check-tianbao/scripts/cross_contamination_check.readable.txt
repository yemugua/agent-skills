def norm(k):
    return str(k or '').strip().lower().replace(' ', '')


def order_check(steps):
    """steps: 已完成的步骤列表(按实际执行顺序)。返回判定与违规。"""
    if steps is None:
        steps = []
    if not isinstance(steps, list):
        raise ValueError('steps 必须是列表')
    # 保留用户提交的实际执行顺序（去重但不改顺序），不能按 CORRECT_ORDER 重排，
    # 否则无论用户怎么填，顺序都会被规范化成正序，错误顺序永远检测不出来。
    order = []
    for s in steps:
        n = norm(s)
        if n not in CORRECT_ORDER:
            raise ValueError('未知的步骤: %s，可选 %s 之一' % (s, '/'.join(CORRECT_ORDER)))
        if n not in order:
            order.append(n)
    good = order == CORRECT_ORDER
    return order, good


def check_practices(flags):
    if flags is None:
        flags = {}
    if not isinstance(flags, dict):
        raise ValueError('practices 必须是对象')
    for k in flags:
        if not any(norm(c[0]) == norm(k) for c in ORDER_CHECKS):
            raise ValueError('未知的检查项: %s，可选项见 ORDER_CHECKS 表' % k)
    rows = []
    bad = []
    unk = []
    for key, name, desc in ORDER_CHECKS:
        v = flags.get(key)
        if v is None:
            rows.append({'name': name, 'status': '未检查', 'desc': desc})
            unk.append(name)
        elif v:
            rows.append({'name': name, 'status': '合格', 'desc': desc})
        else:
            rows.append({'name': name, 'status': '不合格', 'desc': desc})
            bad.append(name)
    return rows, bad, unk


def calc(data):
    steps = data.get('steps') or []
    order, order_ok = order_check(steps)
    rows_p, bad_p, unk_p = check_practices(data.get('practices'))

    fridge_t = data.get('fridge_temp')
    freezer_t = data.get('freezer_temp')
    temp_rows = []
    fridge_ok = None
    freezer_ok = None
    if fridge_t is not None:
        v = float(fridge_t)
        fridge_ok = v <= TEMP['fridge_max']
        temp_rows.append({'name': '冷藏室', 'value': v, 'limit': TEMP['fridge_max'],
                          'status': '达标' if fridge_ok else '超标',
                          'note': '不高于 4 摄氏度为达标'})
    if freezer_t is not None:
        v = float(freezer_t)
        freezer_ok = v <= TEMP['freezer_max']
        temp_rows.append({'name': '冷冻室', 'value': v, 'limit': TEMP['freezer_max'],
                          'status': '达标' if freezer_ok else '超标',
                          'note': '不高于负 18 摄氏度为达标'})

    # 判定顺序: 先看有没有真实缺陷(顺序错/检查项不合格/温度超标), 再看信息是否完整。
    # 「未填 steps」只是信息缺失, 不能盖过已经查出的不合格项, 否则会出现
    # 「只填了温度超标, 却因为没填步骤而被判成待确认」这种把风险说轻了的情况。
    defect_temp = fridge_ok is False or freezer_ok is False
    if order and not order_ok:
        level = '存在交叉污染风险'
    elif bad_p or defect_temp:
        level = '存在薄弱环节'
    elif not order or unk_p:
        level = '待确认'
    else:
        level = '操作规范'

    warns = []
    if not order:
        warns.append('未填写操作步骤，无法判断顺序，请按先洗菜 -> 再切生肉 -> 最后切熟食核对')
    elif not order_ok:
        if len(order) < len(CORRECT_ORDER):
            warns.append('操作顺序不完整，正确顺序应为：先洗菜 -> 再切生肉 -> 最后切熟食')
        else:
            warns.append('操作顺序错误，正确顺序应为：先洗菜 -> 再切生肉 -> 最后切熟食，'
                         '当前执行为 %s' % ' -> '.join(order))
    for b in bad_p:
        desc = next(c[2] for c in ORDER_CHECKS if c[1] == b)
        warns.append('不合格：%s（%s）' % (b, desc))
    for u in unk_p:
        warns.append('未检查：%s，建议逐项确认' % u)
    if fridge_ok is False:
        warns.append('冷藏室温度超标（%.1f 摄氏度，高于 4 摄氏度），'
                     '细菌在 4 至 60 摄氏度区间会快速繁殖' % float(fridge_t))
    if freezer_ok is False:
        warns.append('冷冻室温度超标（%.1f 摄氏度，高于负 18 摄氏度），冷冻效果下降')

    if bad_p or not order_ok or fridge_ok is False or freezer_ok is False:
        advises = list(PRACTICES)
    else:
        advises = []

    return {
        'level': level,
        'order': order,
        'order_ok': order_ok,
        'practice_rows': rows_p,
        'bad': bad_p,
        'unchecked': unk_p,
        'temp_rows': temp_rows,
        'warns': warns,
        'advises': advises,
        'order_detail': [{'key': k, 'name': n, 'how': h, 'wrong': w}
                         for k, n, h, w in ORDER_DETAIL],
        'boards': [{'color': c, 'use': u, 'note': nt} for c, u, nt in BOARD_ZONES],
        'pathogens': [{'key': k, 'name': n, 'source': s, 'ignore': i}
                      for k, n, s, i in PATHOGENS],
        'high_risk': [{'group': g, 'note': nt} for g, nt in HIGH_RISK],
        'banned': BANNED,
    }

def selftest():
    errs = []
    n = 0

    def ck(cond, msg):
        nonlocal n
        n += 1
        if not cond:
            errs.append(msg)

    # 1 正确顺序
    r1 = calc({'steps': ['wash_veg', 'cut_raw_meat', 'cut_cooked']})
    ck(r1['order_ok'] is True, '先洗菜->切生肉->切熟食应正确，实为 %s' % r1['order_ok'])
    ck(r1['level'] == '待确认', '顺序对但未填检查项应为待确认，实为 %s' % r1['level'])

    # 2 错误顺序: 先切生肉后洗菜
    r2 = calc({'steps': ['cut_raw_meat', 'wash_veg', 'cut_cooked']})
    ck(r2['order_ok'] is False, '先切生肉再洗菜应判错误，实为 %s' % r2['order_ok'])
    ck(r2['level'] == '存在交叉污染风险', '顺序错误应判风险，实为 %s' % r2['level'])
    ck(any('操作顺序错误' in w for w in r2['warns']), '应报顺序错误，实为 %s' % r2['warns'])

    # 3 步骤不完整
    r3 = calc({'steps': ['cut_cooked', 'wash_veg']})
    ck(r3['order_ok'] is False, '缺步骤应判顺序不完整，实为 %s' % r3['order_ok'])
    ck(any('不完整' in w for w in r3['warns']), '应报顺序不完整，实为 %s' % r3['warns'])

    # 4 全合格
    allok = {k: True for k, _, _ in ORDER_CHECKS}
    r4 = calc({'steps': ['wash_veg', 'cut_raw_meat', 'cut_cooked'], 'practices': allok,
               'fridge_temp': 3, 'freezer_temp': -20})
    ck(r4['level'] == '操作规范', '全部合格应判操作规范，实为 %s' % r4['level'])
    ck(len(r4['temp_rows']) == 2, '温度应两行，实为 %d' % len(r4['temp_rows']))
    ck(all(t['status'] == '达标' for t in r4['temp_rows']), '温度应全部达标')

    # 5 冷藏超标
    r5 = calc({'practices': allok, 'fridge_temp': 8})
    ck(r5['level'] == '存在薄弱环节', '冷藏超标应判薄弱环节，实为 %s' % r5['level'])
    ck(any('冷藏室温度超标' in w for w in r5['warns']), '应报冷藏超标，实为 %s' % r5['warns'])

    # 6 温度边界: 4 度达标, 4.1 超标
    ck(calc({'fridge_temp': 4})['temp_rows'][0]['status'] == '达标', '4 摄氏度应达标')
    ck(calc({'fridge_temp': 4.5})['temp_rows'][0]['status'] == '超标', '4.5 摄氏度应超标')
    ck(calc({'freezer_temp': -18})['temp_rows'][0]['status'] == '达标', '负 18 摄氏度应达标')
    ck(calc({'freezer_temp': -15})['temp_rows'][0]['status'] == '超标', '负 15 摄氏度应超标')

    # 7 检查项不合格
    r7 = calc({'practices': dict(allok, hand_washed=False)})
    ck(r7['level'] == '存在薄弱环节', '洗手不合格应判薄弱环节，实为 %s' % r7['level'])
    ck('处理生肉后洗手' in r7['bad'], '应点名洗手项，实为 %s' % r7['bad'])
    ck(len(r7['advises']) == len(PRACTICES), '有风险时应给出全部做法建议')

    # 8 致病菌要点
    pnames = [p[1] for p in PATHOGENS]
    for nm in ('沙门氏菌', '李斯特菌', '弯曲菌'):
        ck(nm in pnames, '必须含%s' % nm)
    ck(any('加热不能破坏' in p[3] for p in PATHOGENS), '必须含耐热毒素说明')
    ck(any('仅冲洗不能杀灭' in p[3] for p in PATHOGENS), '必须含冲洗不能杀灭的说明')

    # 9 砧板分区与生熟做法
    ck(len(BOARD_ZONES) >= 5, '砧板分区至少 5 档，实为 %d' % len(BOARD_ZONES))
    ck(any('按颜色分区' in p for p in PRACTICES), '做法必须含按颜色分区')
    ck(any('用皂液洗手' in p for p in PRACTICES), '做法必须含洗手')
    ck(any('底部或专用密闭容器' in p for p in PRACTICES), '做法必须含冰箱隔离存放')

    # 10 高风险人群
    hgroups = [h[0] for h in HIGH_RISK]
    for g in ('婴幼儿', '孕妇', '65 岁以上老人', '免疫力低下者'):
        ck(g in hgroups, '高风险人群必须含%s' % g)

    # 11 禁止项
    ck(len(BANNED) >= 8, '禁止项至少 8 条，实为 %d' % len(BANNED))

    # 12 BOM 两条路径
    bom = '﻿{"steps": ["wash_veg", "cut_cooked"], "fridge_temp": 6}'
    r12 = calc(json.loads(bom.lstrip('﻿')))
    ck(r12['order_ok'] is False, 'BOM 输入应正确解析且顺序不完整，实为 %s' % r12['order_ok'])
    ck(r12['temp_rows'][0]['value'] == 6.0, '温度应解析为 6，实为 %s' % r12['temp_rows'][0]['value'])
    tmp = os.path.join(os.path.dirname(os.path.abspath(__file__)), '_bom_probe8.json')
    with open(tmp, 'w', encoding='utf-8-sig') as f:
        f.write(json.dumps({'steps': ['cut_cooked', 'cut_raw_meat', 'wash_veg']}))
    try:
        r12b = calc(json.loads(open(tmp, 'r', encoding='utf-8-sig').read()))
        ck(r12b['order_ok'] is False, 'utf-8-sig 读取应正常，反序应判错，实为 %s' % r12b['order_ok'])
        ck(r12b['order'][0] == 'cut_cooked', '应保留实际执行顺序，实为 %s' % r12b['order'])
    finally:
        if os.path.exists(tmp):
            os.remove(tmp)

    # 13 非法输入
    for bad, why in (
        ({'steps': ['不存在的步骤']}, '未知步骤'),
        ({'steps': '不是列表'}, 'steps 非列表'),
        ({'practices': '不是对象'}, 'practices 非对象'),
        ({'practices': {'不存在的项': True}}, '未知检查项'),
    ):
        try:
            calc(bad)
            ck(False, '用例13 %s 应抛错' % why)
        except ValueError:
            ck(True, '')

    # 14 输出结构
    md = render_md(r2)
    for key in ('操作顺序', '砧板颜色分区', '常见致病菌', '高风险人群警示',
                '生熟分开的具体做法', '明确禁止', '人工确认清单'):
        ck(key in md, 'Markdown 输出应含章节 %s' % key)
    ck(len(r2['practice_rows']) == 8, '检查项应 8 条，实为 %d' % len(r2['practice_rows']))

    out = os.path.join(os.path.dirname(os.path.abspath(__file__)), '_selftest_result.txt')
    msg = 'SELFTEST FAIL\n' + '\n'.join(errs) if errs else (
        'SELFTEST OK\n断言数: %d\n覆盖: 正确顺序/顺序错误/步骤不完整/全合格/冷藏超标/'
        '温度边界值(4 与 4.5、负 18 与负 15)/检查项不合格/致病菌要点/砧板分区/'
        '生熟做法/高风险人群/禁止项/BOM 两条路径/非法输入/输出结构' % n)
    with open(out, 'w', encoding='utf-8') as f:
        f.write(msg + '\n')
    print(msg)
    return 0 if not errs else 1


def main():
    ap = argparse.ArgumentParser(description='食物交叉污染与生熟分开核查')
    ap.add_argument('--file', help='输入 JSON 文件')
    ap.add_argument('--text', help='输入 JSON 字符串')
    ap.add_argument('--out', help='结果输出路径')
    ap.add_argument('--format', default='md', choices=['md', 'json'])
    ap.add_argument('--selftest', action='store_true')
    a = ap.parse_args()

    if a.selftest:
        return selftest()

    raw = a.text or ''
    if a.file:
        if not os.path.exists(a.file):
            print('ERROR --file 不存在: %s' % a.file)
            return 2
        raw = open(a.file, 'r', encoding='utf-8-sig', errors='replace').read()
    if not raw.strip():
        print('ERROR 需要 --file 或 --text')
        return 2
    try:
        data = json.loads(raw.lstrip('﻿'))
    except ValueError as e:
        print('ERROR JSON 解析失败: %s' % e)
        return 2
    try:
        r = calc(data)
    except ValueError as e:
        print('ERROR %s' % e)
        return 2

    out = render_md(r) if a.format == 'md' else json.dumps(r, ensure_ascii=False, indent=2)
    d = os.path.dirname(os.path.abspath(__file__))
    target = a.out or os.path.join(d, '_last_result.txt')
    with open(target, 'w', encoding='utf-8') as f:
        f.write(out + '\n')
    print('结果已写入: %s' % target)
    print(out)
    return 0


if __name__ == '__main__':
    sys.exit(main())
