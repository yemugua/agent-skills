# -*- coding: utf-8 -*-
"""唱歌音准偏差统计与跑调归因工具。

用法:
  python singing_pitch_check.py --file case.json [--json]
  python singing_pitch_check.py --selftest

判定顺序: 先看真实缺陷(明显跑调、音程方向错误、系统性偏移超阈、曲目越界),
再看信息是否完整; 样本不足只记为待确认, 不掩盖已查出的偏差。
"""
import argparse
import io
import json
import os
import sys
import tempfile

HIT_CENTS = 20.0
LEVEL_STABLE = 20.0
LEVEL_BASIC = 35.0
LEVEL_OFF = 60.0
SYS_SHIFT = 25.0
SYS_SHIFT_BAD = 40.0


def mae(devs):
    if not devs:
        return None
    return sum(abs(float(d)) for d in devs) / len(devs)


def hit_rate(devs):
    if not devs:
        return None
    n = sum(1 for d in devs if abs(float(d)) <= HIT_CENTS)
    return float(n) / len(devs)


def signed_mean(devs):
    if not devs:
        return None
    return sum(float(d) for d in devs) / len(devs)


def direction_errors(notes):
    """按输入原序比较目标走向与实际走向, 不做任何重排。"""
    err = 0
    bad = []
    for i in range(1, len(notes)):
        t0 = float(notes[i - 1].get("target_midi", 0))
        t1 = float(notes[i].get("target_midi", 0))
        if t1 == t0:
            continue
        s0 = t0 * 100.0 + float(notes[i - 1].get("dev_cents", 0) or 0)
        s1 = t1 * 100.0 + float(notes[i].get("dev_cents", 0) or 0)
        if (t1 > t0 and s1 <= s0) or (t1 < t0 and s1 >= s0):
            err += 1
            bad.append((i, i + 1))
    return err, bad


def level_of(v):
    if v is None:
        return u"待确认"
    if v <= LEVEL_STABLE:
        return u"音准稳定"
    if v <= LEVEL_BASIC:
        return u"基本稳定"
    if v <= LEVEL_OFF:
        return u"明显偏离"
    return u"明显跑调"


def shift_verdict(m):
    if m is None:
        return u"待确认", u"未提供逐音偏差"
    if m > SYS_SHIFT_BAD:
        return u"不合格", u"系统性偏高（均值 %+.1f 音分），多为喉部紧张或用喊的方式找高音" % m
    if m < -SYS_SHIFT_BAD:
        return u"不合格", u"系统性偏低（均值 %+.1f 音分），多为气息支撑不足、句尾掉气或音域不够高" % m
    if m > SYS_SHIFT:
        return u"需改进", u"整体偏高 %+.1f 音分，先放松喉部再逐音纠偏" % m
    if m < -SYS_SHIFT:
        return u"需改进", u"整体偏低 %+.1f 音分，先练气息支撑和句尾保持" % m
    return u"合格", u"无系统性偏移（均值 %+.1f 音分）" % m


def range_verdict(range_low, range_high, song_low, song_high):
    if range_high is None or range_low is None or song_high is None or song_low is None:
        return u"待确认", u"音域或曲目音域未提供，无法判断是否越界"
    if float(song_high) > float(range_high):
        gap = float(song_high) - float(range_high)
        return u"不合格", u"曲目最高音超出舒适最高音 %g 个半音，越界的高音怎么练都上不去，先换调" % gap
    if float(song_low) < float(range_low):
        gap = float(range_low) - float(song_low)
        return u"不合格", u"曲目最低音低于舒适最低音 %g 个半音，低音区会虚且跑，先换调" % gap
    return u"合格", u"曲目音域在舒适音域内"


def load_json(path):
    with io.open(path, "r", encoding="utf-8-sig") as f:
        return json.load(f)


def build_report(data):
    notes = data.get("notes", []) or []
    devs = [float(n.get("dev_cents", 0) or 0) for n in notes]
    n = len(notes)
    v = mae(devs)
    hr = hit_rate(devs)
    sm = signed_mean(devs)
    lv = level_of(v)
    err, bad = direction_errors(notes)
    rv, rmsg = range_verdict(
        data.get("range_low"), data.get("range_high"),
        data.get("song_low"), data.get("song_high"))
    breath = data.get("breath_support", "ok")
    monitor = data.get("monitor_able", True)
    tempo = data.get("tempo_bpm", 80)

    defects = []
    fine = []
    pending = []

    if n == 0:
        pending.append(u"未提供逐音偏差，无法统计")
    elif n < 4:
        pending.append(u"仅 %d 个音，样本不足，结论仅供参考" % n)

    if lv == u"明显跑调":
        defects.append(u"平均绝对偏差 %.1f 音分，达到明显跑调，先从单音模仿练起，不要练整首歌" % v)
    elif lv == u"明显偏离":
        fine.append(u"平均绝对偏差 %.1f 音分，听众能听出跑调，先做系统性纠偏" % v)
    if hr is not None and hr < 0.8:
        fine.append(u"命中率仅 %.0f%%，需把逐音偏差压到 20 音分以内" % (hr * 100))

    sv, smsg = shift_verdict(sm)
    if sv == u"不合格":
        defects.append(smsg)
    elif sv == u"需改进":
        fine.append(smsg)
    elif sv == u"待确认":
        pending.append(smsg)

    if err > 0:
        defects.append(u"音程方向错误 %d 处（第 %s 个音之间），回到二音模进单独练跳进" % (
            err, u"、".join(u"%d-%d" % b for b in bad)))

    if rv == u"不合格":
        defects.append(rmsg)
    elif rv == u"待确认":
        pending.append(rmsg)

    if not monitor:
        fine.append(u"听不到自己的声音，这是清唱还行、跟伴奏就跑的首要原因：压低伴奏音量，耳机只戴一只耳朵")
    if breath == "weak":
        fine.append(u"气息支撑不足：先练吸足后匀速吐气数到 20，再谈音准")
    if float(tempo or 0) > 100.0 and v is not None and v > LEVEL_BASIC:
        fine.append(u"速度 %g BPM 偏快且偏差已达 %.1f 音分，先降速到 70 BPM 练准再提速" % (float(tempo), v))

    lines = []
    lines.append("=" * 62)
    lines.append(u"唱歌音准偏差统计与跑调归因")
    lines.append("=" * 62)
    lines.append(u"样本音数：%d    伴奏速度：%g BPM" % (n, float(tempo or 0)))
    lines.append("")
    lines.append(u"[一] 偏差统计")
    lines.append(u"%-18s %s" % (u"平均绝对偏差", (u"%.1f 音分" % v) if v is not None else u"--"))
    lines.append(u"%-18s %s" % (u"命中率", (u"%.0f%%" % (hr * 100)) if hr is not None else u"--"))
    lines.append(u"%-18s %s" % (u"平均有符号偏差", (u"%+.1f 音分" % sm) if sm is not None else u"--"))
    lines.append(u"%-18s %s" % (u"音程方向错误", u"%d 处" % err))
    lines.append(u"%-18s %s" % (u"音准等级", lv))
    lines.append(u"%-18s %s" % (u"系统性偏移判定", sv))
    lines.append(u"%-18s %s" % (u"音域判定", rv))
    lines.append("")
    lines.append(u"[二] 不合格项与原因")
    if defects:
        for i, d in enumerate(defects, 1):
            lines.append(u"  %d. %s" % (i, d))
    else:
        lines.append(u"  无")
    lines.append("")
    lines.append(u"[三] 需改进项")
    if fine:
        for i, d in enumerate(fine, 1):
            lines.append(u"  %d. %s" % (i, d))
    else:
        lines.append(u"  无")
    lines.append("")
    lines.append(u"[四] 待确认项")
    if pending:
        for i, d in enumerate(pending, 1):
            lines.append(u"  %d. %s" % (i, d))
    else:
        lines.append(u"  无")
    lines.append("")
    lines.append(u"[五] 训练处方")
    if lv == u"明显跑调":
        lines.append(u"  1. 阶段一：单音模仿，连续 8 个音偏差 ≤ 20 音分再进阶")
    elif lv == u"明显偏离":
        lines.append(u"  1. 阶段二：二音模进，先解决系统性偏移")
    elif lv == u"基本稳定":
        lines.append(u"  1. 阶段三：一个八度音阶慢速上下行，压长音与跳进")
    else:
        lines.append(u"  1. 阶段四至五：短句与带伴奏，命中率维持 80%% 以上")
    if err > 0:
        lines.append(u"  2. 跳进单独练：先把方向唱对，再谈偏差大小")
    if not monitor:
        lines.append(u"  3. 改监听：伴奏压到人声之下，耳机只戴一只耳朵")
    if breath == "weak":
        lines.append(u"  4. 气息练习每日 5 分钟：吸足后匀速吐气数到 20")
    lines.append(u"  5. 每轮训练录音回听，只标偏高偏低方向错，不做情绪评价")
    lines.append(u"  6. 每两周用同一首测试曲复评，看平均绝对偏差是否下降 10 音分以上")

    if defects:
        overall = u"不合格"
    elif fine:
        overall = u"需改进"
    elif pending:
        overall = u"待确认"
    else:
        overall = u"合格"

    lines.append("")
    lines.append(u"总判定：%s" % overall)
    lines.append(u"提示：声嘶超过两周、听力下降、胸闷气短时停止训练并就医。")
    lines.append(u"免责：本结果为音准自查输出，不构成医学诊断或声乐水平评价。")
    lines.append("=" * 62)

    summary = {
        "note_count": n,
        "mae": v,
        "hit_rate": hr,
        "signed_mean": sm,
        "level": lv,
        "direction_errors": err,
        "shift_verdict": sv,
        "range_verdict": rv,
        "defects": defects,
        "fine": fine,
        "pending": pending,
        "overall": overall,
    }
    return "\n".join(lines), summary


def _run_checks():
    r = []
    r.append((u"平均绝对偏差计算", abs(mae([10.0, -10.0, 20.0]) - 13.3333) < 1e-3))
    r.append((u"命中率计算", abs(hit_rate([10.0, 30.0, -50.0]) - 1.0 / 3.0) < 1e-9))
    r.append((u"有符号均值计算", abs(signed_mean([-10.0, -30.0]) + 20.0) < 1e-9))
    r.append((u"空样本返回 None", mae([]) is None and hit_rate([]) is None))
    r.append((u"等级映射", level_of(15.0) == u"音准稳定" and level_of(70.0) == u"明显跑调"))

    e, _ = direction_errors([{"target_midi": 60, "dev_cents": 0},
                             {"target_midi": 62, "dev_cents": 0}])
    r.append((u"同向音程无方向错误", e == 0))
    e2, _ = direction_errors([{"target_midi": 60, "dev_cents": 0},
                              {"target_midi": 61, "dev_cents": -150}])
    r.append((u"反例：目标上行实际下行判方向错误", e2 == 1))
    e3, _ = direction_errors([{"target_midi": 61, "dev_cents": -150},
                              {"target_midi": 60, "dev_cents": 0}])
    r.append((u"反例：目标下行实际上行判方向错误", e3 == 1))

    _, s = build_report({
        "notes": [{"target_midi": 60 + i, "dev_cents": 8 * (i % 3) - 8} for i in range(8)],
        "range_low": 55, "range_high": 72, "song_low": 57, "song_high": 70,
        "breath_support": "ok", "monitor_able": True, "tempo_bpm": 80,
    })
    r.append((u"正常样例判定合格", s["overall"] == u"合格"))

    _, s1 = build_report({
        "notes": [{"target_midi": 60, "dev_cents": -80} for _ in range(8)],
        "range_low": 55, "range_high": 72, "song_low": 57, "song_high": 70,
        "breath_support": "ok", "monitor_able": True, "tempo_bpm": 80,
    })
    r.append((u"反例：整体偏低 80 音分判不合格", s1["overall"] == u"不合格"))
    r.append((u"反例：明显跑调进缺陷表", any(u"明显跑调" in d for d in s1["defects"])))

    _, s2 = build_report({
        "notes": [{"target_midi": 60, "dev_cents": 0},
                  {"target_midi": 61, "dev_cents": -150},
                  {"target_midi": 62, "dev_cents": 0},
                  {"target_midi": 63, "dev_cents": 0}],
        "range_low": 55, "range_high": 72, "song_low": 57, "song_high": 70,
        "breath_support": "ok", "monitor_able": True, "tempo_bpm": 80,
    })
    r.append((u"反例：方向错误判不合格", s2["overall"] == u"不合格"))
    r.append((u"反例：方向错误进缺陷表", any(u"音程方向错误" in d for d in s2["defects"])))

    _, s3 = build_report({
        "notes": [{"target_midi": 60 + i, "dev_cents": 5} for i in range(8)],
        "range_low": 55, "range_high": 72, "song_low": 57, "song_high": 78,
        "breath_support": "ok", "monitor_able": True, "tempo_bpm": 80,
    })
    r.append((u"反例：曲目超出音域判不合格", s3["overall"] == u"不合格"))
    r.append((u"反例：音域越界进缺陷表", any(u"换调" in d for d in s3["defects"])))

    _, s4 = build_report({
        "notes": [{"target_midi": 60, "dev_cents": -95},
                  {"target_midi": 61, "dev_cents": -90}],
        "range_low": 55, "range_high": 72, "song_low": 57, "song_high": 70,
        "breath_support": "ok", "monitor_able": True, "tempo_bpm": 80,
    })
    r.append((u"反例：样本不足但偏差巨大不得判待确认", s4["overall"] == u"不合格"))

    _, s5 = build_report({"notes": []})
    r.append((u"无逐音数据记为待确认", s5["overall"] == u"待确认"))

    _, s6 = build_report({
        "notes": [{"target_midi": 60 + i, "dev_cents": 5} for i in range(8)],
        "range_low": 55, "range_high": 72, "song_low": 57, "song_high": 70,
        "breath_support": "weak", "monitor_able": False, "tempo_bpm": 80,
    })
    r.append((u"监听与气息问题判需改进", s6["overall"] == u"需改进"))
    r.append((u"监听问题进需改进表", any(u"听不到自己" in d for d in s6["fine"])))

    p = os.path.join(tempfile.gettempdir(), "_singing_bom_case.json")
    blob = b"\xef\xbb\xbf" + json.dumps({
        "notes": [{"target_midi": 60, "dev_cents": 5}],
        "range_low": 55, "range_high": 72, "song_low": 57, "song_high": 70,
    }, ensure_ascii=False).encode("utf-8")
    with open(p, "wb") as f:
        f.write(blob)
    try:
        d = load_json(p)
        r.append((u"UTF-8 BOM 输入首键可读", len(d.get("notes", [])) == 1))
        r.append((u"UTF-8 BOM 输入数值可读", abs(d["notes"][0]["dev_cents"] - 5.0) < 1e-9))
    finally:
        if os.path.exists(p):
            os.remove(p)

    return r


def selftest():
    results = _run_checks()
    failed = [name for name, ok in results if not ok]
    ok = len(failed) == 0
    lines = ["SELFTEST %s" % ("OK" if ok else "FAILED")]
    lines.append("checks=%d passed=%d failed=%d" % (len(results), len(results) - len(failed), len(failed)))
    for name in failed:
        lines.append("FAILED: %s" % name)
    with io.open(os.path.join(os.path.dirname(os.path.abspath(__file__)), "_selftest_result.txt"),
                 "w", encoding="utf-8") as f:
        f.write("\n".join(lines))
    print("\n".join(lines))
    return ok


def main():
    parser = argparse.ArgumentParser(description=u"唱歌音准偏差统计工具")
    parser.add_argument("--file", help=u"输入 JSON 文件路径（UTF-8）")
    parser.add_argument("--json", action="store_true", help=u"以 JSON 输出")
    parser.add_argument("--out", help=u"结果写入指定文件")
    parser.add_argument("--selftest", action="store_true", help=u"运行自检")
    args = parser.parse_args()

    here = os.path.dirname(os.path.abspath(__file__))

    if args.selftest:
        sys.exit(0 if selftest() else 1)

    if not args.file:
        parser.error(u"需要 --file 或 --selftest")

    data = load_json(args.file)

    report, summary = build_report(data)
    body = json.dumps(summary, ensure_ascii=False, indent=2) if args.json else report
    out_path = args.out or os.path.join(here, "_last_result.txt")
    with io.open(out_path, "w", encoding="utf-8") as f:
        f.write(body)
    print(body)
    print("")
    print(u"结果已写入：%s" % out_path)


if __name__ == "__main__":
    main()
